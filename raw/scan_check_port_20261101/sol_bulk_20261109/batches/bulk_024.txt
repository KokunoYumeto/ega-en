\begin{proposition}[6.6.1]
\label{IV.6.6.1}
For a locally Noetherian prescheme $\mathrm{Z}$, let us denote by $\boldsymbol{P}(\mathrm{Z})$ any one of the following properties:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\mathrm{Z}$ is a Cohen-Macaulay prescheme.
  \item $\mathrm{Z}$ satisfies $(\mathrm{S}_n)$.
  \item $\mathrm{Z}$ is regular.
  \item $\mathrm{Z}$ satisfies $(\mathrm{R}_n)$.
  \item $\mathrm{Z}$ is normal.
  \item $\mathrm{Z}$ is reduced.
\end{enumerate}
Let then $\mathrm{X}$, $\mathrm{Y}$, $\mathrm{Z}$ be three locally Noetherian preschemes, $f:\mathrm{X}\to\mathrm{Y}$, $g:\mathrm{Y}\to\mathrm{Z}$ two morphisms.
\begin{enumerate}[label=(\roman*)]
  \item Suppose that $f$ and $g$ are flat and that for every $y\in f(\mathrm{X})$ \textnormal{(}resp.\ every $z\in g(\mathrm{Y})$\textnormal{)}, the prescheme $f^{-1}(y)$ \textnormal{(}resp.\ $g^{-1}(z)$\textnormal{)} satisfies $\boldsymbol{P}$. Then $h=g\circ f$ is flat and, for every $z\in h(\mathrm{X})$, $h^{-1}(z)$ satisfies $\boldsymbol{P}$.

  \item Suppose the following conditions are satisfied: $f$ is faithfully flat, $h=g\circ f$ is flat, for every $y\in\mathrm{Y}$, the prescheme $f^{-1}(y)$ satisfies $\boldsymbol{P}$, and for every $z\in h(\mathrm{X})$, the prescheme $h^{-1}(z)$ satisfies $\boldsymbol{P}$. Then $g$ is flat, and for every $z\in g(\mathrm{Y})$, $g^{-1}(z)$ satisfies $\boldsymbol{P}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) We already know \sref{IV.2.1.6} that $h$ is flat; furthermore, for every $z\in\mathrm{Z}$, $f_z=f\otimes 1_{\boldsymbol{k}(z)}:h^{-1}(z)\to g^{-1}(z)$ is flat \sref{IV.2.1.4}, and for every $y\in g^{-1}(z)$, $f_z^{-1}(y)$ is isomorphic to $f^{-1}(y)$ by transitivity of fibres \sref[I]{I.3.6.4}. The conclusion follows then respectively from \sref{IV.6.3.5},~(ii), \sref{IV.6.4.2}, \sref{IV.6.5.2},~(ii), \sref{IV.6.5.3},~(ii), \sref{IV.6.5.4},~(ii) and \sref{IV.3.3.5},~(ii).

(ii) We already know that $g$ is flat \sref{IV.2.2.13}; in addition, for every $z\in\mathrm{Z}$, $f_z$ is faithfully flat \sref{IV.2.2.13}. The conclusion follows then respectively from \sref{IV.6.3.5},~(i), \sref{IV.6.4.2}, \sref{IV.6.5.2},~(i), \sref{IV.6.5.3},~(i), \sref{IV.6.5.4},~(i) and \sref{IV.2.1.13}.
\end{proof}

\begin{remark}[6.6.2]
\label{IV.6.6.2}
Suppose $h$ flat, $f$ faithfully flat, and suppose that for every $z\in\mathrm{Z}$, $h^{-1}(z)$ is \emph{of coprofundity} $\leqslant n$ \sref{IV.5.7.1}; then it follows from the reasoning of \sref{IV.6.6.1},~(ii) and \sref{IV.6.3.2} that $g^{-1}(z)$ is \emph{of coprofundity} $\leqslant n$ for every $z\in g(\mathrm{Y})$ and that for every $y\in\mathrm{Y}$, $f^{-1}(y)$ is \emph{of coprofundity} $\leqslant n$.
\end{remark}

\begin{proposition}[6.7.1]
\label{IV.6.7.1}
Let $k$ be a field, $\mathrm{X}$ a $k$-prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_\mathrm{X}$-Module coherent. Let $k'$ be an extension of $k$; set $\mathrm{X}'=\mathrm{X}\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$ and let $p:\mathrm{X}'\to\mathrm{X}$ be the canonical projection. We suppose either that $\mathrm{X}$ is locally of finite type over $k$, or that $k'$ is an extension of finite type of $k$, so that in both cases $\mathrm{X}'$ is locally Noetherian. Let $x'$ be a point of $\mathrm{X}'$, $x=p(x')$. Then:
\begin{enumerate}[label=(\roman*)]
  \item We have $\operatorname{coprof}(\sh{F}_x)=\operatorname{coprof}(\sh{F}'_{x'})$; in particular, for $\sh{F}_x$ to be an $\sh{O}_x$-module of Cohen-Macaulay, it is necessary and sufficient that $\sh{F}'_{x'}$ be an $\sh{O}_{x'}$-module of Cohen-Macaulay.
  \item For $\sh{F}$ to satisfy the property $(\mathrm{S}_n)$ at the point $x$, it is necessary and sufficient that $\sh{F}'$ satisfy $(\mathrm{S}_n)$ at the point $x'$.
\end{enumerate}
\end{proposition}

\begin{lemma}[6.7.1.1]
\label{IV.6.7.1.1}
Let $\mathrm{K}$, $\mathrm{L}$ be two extensions of a field $k$, one of which is of finite type (so that the ring $\mathrm{K}\otimes_k\mathrm{L}$ is Noetherian). Then $\mathrm{K}\otimes_k\mathrm{L}$ is a Cohen-Macaulay ring.
\end{lemma}

\begin{proof}
Suppose for example that $\mathrm{L}$ is an extension of finite type of $k$, so that $\mathrm{L}$ is a finite extension of a pure extension $k(\mathbf{t})$ of $k$ ($\mathbf{t}$ being a system $(t_i)_{1\leqslant i\leqslant m}$ of indeterminates). If one sets $\mathrm{A}=\mathrm{K}\otimes_k k(\mathbf{t})$, $\mathrm{B}=\mathrm{K}\otimes_k\mathrm{L}$, $\mathrm{B}$ is a flat $\mathrm{A}$-module \sref[0\textsubscript{\mdseries I}]{0.6.2.1} and of finite type; the morphism $h:\operatorname{Spec}(\mathrm{B})\to\operatorname{Spec}(\mathrm{A})$ is therefore finite, and as every Artinian prescheme is of Cohen-Macaulay, to prove that $\mathrm{B}$ is a Cohen-Macaulay ring, it suffices to prove that $\mathrm{A}$ is a Cohen-Macaulay ring \sref{IV.6.3.5}. Now, if $\mathrm{S}$ is the set of elements $\neq 0$ of $k[\mathbf{t}]$, then $\mathrm{A}=\mathrm{S}^{-1}\mathrm{A}'$, where $\mathrm{A}'=\mathrm{K}\otimes_k k[\mathbf{t}]=\mathrm{K}[\mathbf{t}]$; by \sref[0]{0.16.5.13}, it suffices to prove that $\mathrm{A}'$ is a Cohen-Macaulay ring; but this follows from the fact that $\mathrm{A}'$ is \emph{regular} \sref[0]{0.17.3.7}.
\end{proof}

\begin{corollary}[6.7.2]
\label{IV.6.7.2}
For $\sh{O}_x$ to be a Cohen-Macaulay ring (resp. for $\mathrm{X}$ to satisfy $(\mathrm{S}_n)$ at the point $x$), it is necessary and sufficient that $\sh{O}_{x'}$ be a Cohen-Macaulay ring (resp. that $\mathrm{X}'$ satisfy $(\mathrm{S}_n)$ at the point $x'$).
\end{corollary}

\begin{corollary}[6.7.3]
\label{IV.6.7.3}
Let $\mathrm{X}$ and $\mathrm{Y}$ be two locally Noetherian $k$-preschemes, at least one of which is locally of finite type over $k$. Let $\sh{F}$ (resp. $\sh{G}$) be a coherent $\sh{O}_{\mathrm{X}}$-Module (resp. a coherent $\sh{O}_{\mathrm{Y}}$-Module). If $\sh{F}$ and $\sh{G}$ satisfy property $(\mathrm{S}_n)$, then so does $\sh{F}\otimes_k\sh{G}$.
\end{corollary}

\begin{proof}
The hypothesis implies that $\mathrm{X}\times_k\mathrm{Y}$ is locally Noetherian; let $p:\mathrm{X}\times_k\mathrm{Y}\to\mathrm{X}$ and $q:\mathrm{X}\times_k\mathrm{Y}\to\mathrm{Y}$ be the canonical projections, which are flat morphisms. Suppose, for example, that $\mathrm{X}$ is locally of finite type over $k$; we may write $\sh{F}\otimes_k\sh{G}=p^*(\sh{F})\otimes_k\mathscr G$ and since $\sh{F}$ is flat relative to the structural morphism $\mathrm{X}\to\operatorname{Spec}(k)$, $p^*(\sh{F})$ is $q$-flat \sref{IV.2.1.4}. Applying the criterion \sref{IV.6.4.1},~(ii) to the morphism $q$, it suffices to see that, for every $y\in\mathrm{Y}$, $p^*(\sh{F})\otimes_{\sh{O}_{\mathrm{Y}}}\boldsymbol{k}(y)$ has property $(\mathrm{S}_n)$; but
$p^*(\sh{F})\otimes_{\sh{O}_{\mathrm{Y}}}\boldsymbol{k}(y)=\sh{F}\otimes_k\boldsymbol{k}(y)$, and since $\mathrm{X}$ is locally of finite type over $k$, the conclusion follows from \sref{IV.6.7.1},~(ii).
\end{proof}

\begin{lemma}[6.7.4.1]
\label{IV.6.7.4.1}
Let $\mathrm{K}$ and $\mathrm{L}$ be two extensions of a field $k$, one of which is of finite type. If $\mathrm{L}$ is a separable extension of $k$, then the ring $\mathrm{K}\otimes_k\mathrm{L}$ is regular.
\end{lemma}

\begin{proof}
We distinguish two cases.

A) $\mathrm{L}$ is an extension of finite type of $k$. With the notation of \sref{IV.6.7.1.1}, we may then suppose that $\mathrm{L}$ is a finite separable extension of $k(\mathbf{t})$. For every $s\in\operatorname{Spec}(\mathrm{A})$, the ring $\boldsymbol{k}(s)\otimes_{k(\mathbf{t})}\mathrm{L}$ is a direct sum of finitely many fields, finite separable extensions of $\boldsymbol{k}(s)$, and is therefore regular; it follows from \sref{IV.6.5.2},~(ii) that it suffices to prove that $\mathrm{A}$ is regular. Since $\mathrm{A}=\mathrm{S}^{-1}\mathrm{A}'$, it suffices to prove that $\mathrm{A}'$ is regular \sref[0]{0.17.3.6}, and this was proved in the proof of \sref{IV.6.7.1.1}.

B) Let $(\mathrm{L}_\lambda)$ be the filtered family of subextensions of $\mathrm{L}$ that are of finite type over $k$. By A), each ring $\mathrm{C}_\lambda=\mathrm{K}\otimes_k\mathrm{L}_\lambda$ is regular; moreover, if $\mathrm{L}_\lambda\subset\mathrm{L}_\mu$, then $\mathrm{L}_\mu$ is a flat $\mathrm{L}_\lambda$-module, so $\mathrm{C}_\mu$ is a flat $\mathrm{C}_\lambda$-module \sref[0\textsubscript{\mdseries I}]{0.6.2.1}. Since $\mathrm{C}=\mathrm{K}\otimes_k\mathrm{L}=\varinjlim(\mathrm{K}\otimes_k\mathrm{L}_\lambda)$ is Noetherian, we may apply \sref{IV.5.13.7}, and $\mathrm{C}$ is therefore regular.
\end{proof}

\begin{proposition}[6.7.7]
\label{IV.6.7.7}
Let $k$ be a field, $\mathrm{X}$ a $k$-prescheme locally Noetherian, $x$ a point of $\mathrm{X}$; denote by $\boldsymbol{Q}(k')$ the relation ``$k'$ is an extension of $k$, and $\boldsymbol{P}(\mathrm{X}_{(k')},x')$ is true for every point $x'$ of $\mathrm{X}_{(k')}$ whose projection in $\mathrm{X}$ is $x$'', where $\boldsymbol{P}(\mathrm{Z},z)$ is one of the properties \emph{a)}, \emph{b)}, \emph{c)} of the statement \sref{IV.6.7.4}. Then the following properties are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\boldsymbol{Q}(k')$ is true for every finite extension $k'$ of $k$.
  \item $\boldsymbol{Q}(k')$ is true for every finite radicial extension $k'$ of $k$.
  \item $\boldsymbol{Q}(k')$ is true for every extension of finite type $k'$ of $k$.
\end{enumerate}
Suppose in addition that $\mathrm{X}$ is locally of finite type over $k$; then the three preceding properties are also equivalent to the following:
\begin{enumerate}[label=\emph{\alph*)}]
\setcounter{enumi}{3}
  \item $\boldsymbol{Q}(k')$ is true for every extension $k'$ of $k$.
  \item $\boldsymbol{Q}(k')$ is true for a perfect extension $k'$ of $k$.
  \item $\boldsymbol{Q}(k')$ is true for every extension $k'=k^{p^{-s}}$, where $p$ is the characteristic exponent of $k$, and $s>0$.
\end{enumerate}
\end{proposition}

\begin{proof}
To prove the equivalence of $a)$, $b)$ and $c)$, it clearly suffices to show that $b)$ implies $c)$. Let $\mathrm{K}$ be an extension of finite type of $k$, which is then the field of fractions of a $k$-algebra of finite type $\mathrm{A}$; set $\mathrm{Y}=\operatorname{Spec}(\mathrm{A})$. \begin{corollary}[6.7.8]
\label{IV.6.7.8}
Let $k$ be a field; for a $k$-prescheme locally Noetherian $\mathrm{X}$ and a point $x\in\mathrm{X}$, denote by $\boldsymbol{P}(\mathrm{X},x)$ any of the following properties:
\begin{enumerate}[label=(\roman*)]
  \item $\operatorname{coprof}(\sh{O}_x)\leqslant n$.
  \item $\sh{O}_x$ is a Cohen-Macaulay ring.
  \item $\mathrm{X}$ satisfies the property $(\mathrm{S}_n)$ at the point $x$.
  \item $\mathrm{X}$ is geometrically regular at the point $x$.
  \item $\mathrm{X}$ has the property $(\mathrm{R}_n)$ geometric at the point $x$.
  \item $\mathrm{X}$ is geometrically normal at the point $x$.
  \item $\mathrm{X}$ is geometrically reduced (i.e.\ separable) at the point $x$.
\end{enumerate}
Let $k'$ be an extension of $k$; one supposes either that $k'$ is of finite type, or that $\mathrm{X}$ is locally of finite type over $k$, so that $\mathrm{X}'=\mathrm{X}\otimes_k k'$ is locally Noetherian. Let $x'$ be a point of $\mathrm{X}'$ whose projection in $\mathrm{X}$ is $x$. Then, for $\boldsymbol{P}(\mathrm{X},x)$ to be true, it is necessary and sufficient that $\boldsymbol{P}(\mathrm{X}',x')$ be so.
\end{corollary}

\begin{proof}
This has already been seen for property (vii) \sref{IV.4.6.11}, and for properties (i), (ii), and (iii) \sref{IV.6.7.1}. For (iv), (v), and (vi), this follows from \sref{IV.6.7.7}: the necessity follows from the equivalence of criteria $a)$ and $c)$, and from the equivalence of $c)$ and $d)$ when $\mathrm{X}$ is locally of finite type over $k$. To prove sufficiency, let $k''$ be a finite radicial extension of $k$; one may regard $k'$ and $k''$ as subextensions of an extension $\mathrm{K}$ of $k$. Set $\mathrm{X}''=\mathrm{X}\otimes_k k''$ and $\mathrm{X}_0=\mathrm{X}\otimes_k\mathrm{K}$, and note that, since $k''$ is radicial over $k$, there is only \emph{one} point $x''$ of $\mathrm{X}''$ above $x$ \sref[I]{I.3.5.7} and \sref[I]{I.3.5.8}. Let $x_0$ be any point of $\mathrm{X}_0$ above $x'$. If $\boldsymbol{P}(\mathrm{X}',x')$ is true, then so is $\boldsymbol{P}(\mathrm{X}_0,x_0)$ by \sref{IV.6.7.7},~$c)$ and $d)$ (if $k'$ is of finite type over $k$, one may suppose the same of $\mathrm{K}$; if $\mathrm{X}$ is locally of finite type over $k$, then $\mathrm{X}'$ is locally of finite type over $k'$). It follows from \sref{IV.6.7.4} that $\boldsymbol{Q}(k'')$ is true (with the notation of \sref{IV.6.7.7}), and hence $\boldsymbol{P}(\mathrm{X},x)$ is true by \sref{IV.6.7.7},~$b)$.
\end{proof}

\end{enumerate}
if $f$ is flat at the point $x$, and if in addition the corresponding property $\boldsymbol{P}(f^{-1}(f(x)),x)$ \textnormal{(}notation of \sref{IV.6.7.8}\textnormal{)} is true.

We say that $f$ is \emph{smooth} at the point $x$ if it is locally of finite presentation in a neighbourhood of $x$ and if it is regular at the point $x$. We say respectively that $f$ is a morphism: of coprofundity $\leqslant n$, of Cohen-Macaulay, $(\mathrm{S}_n)$, regular, $(\mathrm{R}_n)$, normal, reduced, smooth, if it has the corresponding property at every point of $\mathrm{X}$.
\end{definition}

\begin{proposition}[6.8.2]
\label{IV.6.8.2}
Let $\mathrm{X}$, $\mathrm{Y}$ be two locally Noetherian preschemes, $f:\mathrm{X}\to\mathrm{Y}$ a morphism, $x$ a point of $\mathrm{X}$. Denote by $\boldsymbol{M}(f,x)$ one of the properties (i) to (vii) of the definition \sref{IV.6.8.1}, or the property of being smooth at the point $x$. Let $\mathrm{Y}'$ be a locally Noetherian prescheme, $g:\mathrm{Y}'\to\mathrm{Y}$ a morphism, $\mathrm{X}'=\mathrm{X}\times_\mathrm{Y}\mathrm{Y}'$, $f'=f_{(\mathrm{Y}')}:\mathrm{X}'\to\mathrm{Y}'$. Suppose that $f$ or $g$ is locally of finite type. Then, for every $x'\in\mathrm{X}'$ above $x$, the property $\boldsymbol{M}(f,x)$ implies $\boldsymbol{M}(f',x')$.
\end{proposition}

\begin{proof}
Set $y=f(x)$, $y'=f'(x')$; by transitivity of fibres \sref[I]{I.3.6.4}, we have $f'^{-1}(y')=f^{-1}(y)\otimes_{\boldsymbol{k}(y)}\boldsymbol{k}(y')$. Since either $f^{-1}(y)$ is locally of finite type over $\boldsymbol{k}(y)$, or $\boldsymbol{k}(y')$ is of finite type over $\boldsymbol{k}(y)$ \sref[I]{I.6.4.11}, it follows from \sref{IV.6.7.8} that the properties $\boldsymbol{P}(f^{-1}(y),x)$ and $\boldsymbol{P}(f'^{-1}(y'),x')$ are equivalent; furthermore, if $f$ is flat at the point $x$, $f'$ is flat at the point $x'$ \sref{IV.2.1.4}, which proves the proposition, taking into account \sref{IV.1.4.3},~(iii).
\end{proof}

If the theorem is true for each $\mathrm{N}_i$, there exists for each $i$ an $f_i\neq 0$ in $\mathrm{A}$ such that $(\mathrm{N}_i)_{f_i}$ is free over $\mathrm{A}_{f_i}$; setting $f=f_1f_2\ldots f_{q-1}$, it follows that $(\mathrm{N}_i)_f$ is a free $\mathrm{A}_f$-module for $1\leqslant i\leqslant q-1$. But $(\mathrm{N}_i)_f=(\mathrm{M}_i)_f/(\mathrm{M}_{i+1})_f$ \sref[0\textsubscript{\mdseries I}]{0.1.3.2}, and since an extension of free modules is free, we deduce that $\mathrm{M}_f$ is a free $\mathrm{A}_f$-module.

Replacing $\mathrm{B}$ by $\mathrm{B}/\mathfrak{p}$, where $\mathfrak{p}$ is a prime ideal of $\mathrm{B}$, one can therefore reduce to the case $\mathrm{M}=\mathrm{B}$ with $\mathrm{B}$ integral. We then know (Bourbaki, \emph{Alg.\ comm.}, chap.~V, \S~3, n\textsuperscript{o}~1, cor.~1 of th.~1) that there exist $g\neq 0$ in $\mathrm{A}$ and elements $t_i$ ($1\leqslant i\leqslant m$) of $\mathrm{B}$, algebraically independent over $\mathrm{A}$, such that $\mathrm{B}_g$ is integral over $\mathrm{A}_g[t_1,\ldots,t_m]$. One can replace $\mathrm{A}$ by $\mathrm{A}_g$, $\mathrm{B}$ by $\mathrm{B}_g$ and suppose that $\mathrm{B}$ is integral over $\mathrm{C}=\mathrm{A}[t_1,\ldots,t_m]$, hence a $\mathrm{C}$-module of finite type and \emph{without torsion}. We know furthermore \sref{IV.4.1.2} that the dimension of $\operatorname{Spec}(\mathrm{B}\otimes_\mathrm{A}\mathrm{K})$ is equal to $m$, hence $m=n$.

This being so, if $h$ is the rank of the $\mathrm{C}$-module without torsion $\mathrm{B}$, there exists a short exact sequence of $\mathrm{C}$-modules
\[
0\to\mathrm{C}^h\to\mathrm{B}\to\mathrm{M}'\to 0
\]
where $\mathrm{M}'$ is a torsion $\mathrm{C}$-module of finite type; the support of $\mathrm{M}'$ does not therefore contain the generic point of $\operatorname{Spec}(\mathrm{C})$ \sref[I]{I.7.4.6} and hence the support of $\mathrm{M}'\otimes_\mathrm{A}\mathrm{K}$ does not contain the generic point of $\operatorname{Spec}(\mathrm{C}\otimes_\mathrm{A}\mathrm{K})$ \sref[I]{I.9.1.13.1}; one concludes \sref{IV.4.1.2.1} that its dimension is $<n$. By the induction hypothesis, there exists $f\neq 0$ in $\mathrm{A}$ such that $\mathrm{M}'_f$ is a free $\mathrm{A}_f$-module; moreover $\mathrm{C}^h_f$ is a free $\mathrm{A}_f$-module, and therefore so is $\mathrm{B}_f$, which by \sref[0\textsubscript{\mdseries I}]{0.1.3.2} is an extension of free modules.
\end{proof}

\begin{env}[6.10.1]
\label{IV.6.10.1}
Let $\mathrm{X}$ be a locally Noetherian prescheme, $\sh{J}$ a quasi-coherent Ideal of $\sh{O}_\mathrm{X}$, $\mathrm{Y}$ the closed sub-prescheme of $\mathrm{X}$ defined by $\sh{J}$, $j:\mathrm{Y}\to\mathrm{X}$ the canonical injection. For every integer $k\geqslant 0$, the $\sh{O}_\mathrm{X}$-Module $\sh{J}^k\sh{F}/\sh{J}^{k+1}\sh{F}$ is annihilated by $\sh{J}$, hence of the form $j_*(\sh{G}_k)$, where $\sh{G}_k=j^*(\sh{J}^k\sh{F}/\sh{J}^{k+1}\sh{F})=j^*(\sh{J}^k\sh{F})$ is a coherent $\sh{O}_\mathrm{Y}$-Module. \[

\]
