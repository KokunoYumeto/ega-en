\begin{corollary}[17.10.3]
\label{IV.17.10.3}
Let $f:X\to Y$, $g:Y\to Z$ be two smooth morphisms. Then, for every $x\in X$, we have
\begin{equation}
\label{IV.17.10.3.1}
\tag{17.10.3.1}
\dim_x(g\circ f)=\dim_x f+\dim_{f(x)}g.
\end{equation}
\end{corollary}

Indeed, $g\circ f$ is smooth (17.3.3), hence the three $\sh{O}_X$-modules $\varOmega^1_{X/Y}$, $\varOmega^1_{X/Z}$ and $f^*(\varOmega^1_{Y/Z})$ are locally free (17.2.3 and $\mathbf{0}_{\mathrm I}$, 5.4.5); moreover the rank at $x$ of $f^*(\varOmega^1_{Y/Z})$ is equal to the rank at $y=f(x)$ of $\varOmega^1_{Y/Z}$. Equality (17.10.3.1) is therefore a consequence of (17.10.2) and of the exactness of the sequence (17.2.3.1).

\begin{corollary}[17.10.4]
\label{IV.17.10.4}
Let $f:X\to Y$ be a smooth morphism, $X'$ a sub-prescheme of $X$ such that the composite morphism $X'\xrightarrow{j}X\xrightarrow{f}Y$ (where $j$ is the canonical injection) is smooth. Then the conormal sheaf $\sh{N}_{X'/X}$ is a locally free $\sh{O}_{X'}$-module, and for every $x\in X'$, we have
\begin{equation}
\label{IV.17.10.4.1}
\tag{17.10.4.1}
\dim_x f=\dim_x(f\circ j)+\operatorname{rg}_{\sh{O}_{X',x}}(\sh{N}_{X'/X})_x.
\end{equation}
\end{corollary}

Indeed, $\varOmega^1_{X/Y}\otimes_{\sh{O}_X}\sh{O}_{X'}$ and $\varOmega^1_{X'/Y}$ are then locally free and the exact sequence (17.2.5.1) is split in a suitable neighbourhood of each point of $X'$, hence $\sh{N}_{X'/X}$ is locally free (Bourbaki, \emph{Alg. comm.}, chap. II, § 5, n\textsuperscript{o} 2, th. 1), and relation (17.10.4.1) follows immediately from the exactness of the sequence (17.2.5.1).

\begin{theorem}[17.11.1]
\label{IV.17.11.1}
Let $f:Y\to S$ and $h:X\to S$ be two morphisms locally of finite presentation, $g:X\to Y$ an $S$-morphism, $x$ a point of $X$; set $y=g(x)$, $s=f(y)=h(x)$. The following conditions are equivalent:
\begin{enumerate}
  \item[\textup{a)}] $f$ is smooth at the point $y$ and $g$ is smooth at the point $x$.
  \item[\textup{b)}] $g$ and $h$ are smooth at the point $x$.
  \item[c)] $h$ is smooth at the point $x$, and the canonical homomorphism (16.4.18)
  \begin{equation}
  \label{IV.17.11.1.1}
  \tag{17.11.1.1}
  (g^*(\varOmega^1_{Y/S}))_x\to(\varOmega^1_{X/S})_x
  \end{equation}
  is left invertible (in other words, is an isomorphism onto a \emph{direct factor} of $(\varOmega^1_{X/S})_x$).
  \item[c')] $h$ is smooth at the point $x$, and the canonical homomorphism
  \begin{equation}
  \label{IV.17.11.1.2}
  \tag{17.11.1.2}
  ((\varOmega^1_{Y/S})_y\otimes_{\sh{O}_{Y,y}}\boldsymbol{k}(y))\otimes_{\mathbf{k}(y)}\boldsymbol{k}(x)\to(\varOmega^1_{X/S})_x\otimes_{\sh{O}_{X,x}}\boldsymbol{k}(x)
  \end{equation}
  is injective.
\end{enumerate}
Suppose moreover that the homomorphism $\boldsymbol{k}(y)\to k(x)$ is bijective. Then the preceding conditions are also equivalent to the following:
\begin{enumerate}
  \item[\textup{d)}] $h$ is smooth at the point $x$, and the canonical map $\mathrm{T}_{X/S}(x)\to\mathrm{T}_{Y/S}(y)$ from the tangent vector space to $X$ at $x$ to the tangent vector space to $Y$ at $y$ \textup{(16.5.12)} is surjective.
\end{enumerate}
\end{theorem}

That a) implies b) follows trivially from (17.3.3, (ii)); b) implies c) by application of (17.2.3, (ii)). To see that c) is equivalent to c'), note that by virtue of (17.2.3, (i)), $\varOmega^1_{X/S}$ is a locally free $\sh{O}_X$-module of finite type; it then suffices to apply (\textbf{0}, 19.1.12) to the local ring $\sh{O}_{X,x}$ and to the homomorphism (17.11.1.1) of $\sh{O}_{X,x}$-modules of finite type. When $\boldsymbol{k}(x)=k(y)$, the tangent linear map $\mathrm{T}_{X/S}(x)\to\mathrm{T}_{Y/S}(y)$ is the transpose of (17.11.1.2), by (16.5.12), which establishes in this case the equivalence of c') and d).

Since the canonical images of the $d_{B'/A}(T_i)$ in $\Omega^1_{C/A}$ are those of the $d_{B/A}(s_i)$, they form part of a basis of $\Omega^1_{C/A}$. \begin{corollary}[17.11.2]
\label{IV.17.11.2}
Under the general hypotheses of \textup{(17.11.1)}, the following conditions are equivalent:
\begin{enumerate}
  \item[\textup{a)}] $f$ is smooth at the point $y$ and $g$ is étale at the point $x$.
  \item[\textup{b)}] $h$ is smooth at the point $x$ and $g$ is étale at the point $x$.
  \item[\textup{c)}] $h$ is smooth at the point $x$, and the canonical homomorphism
  \[
  (g^*(\varOmega^1_{Y/S}))_x\to(\varOmega^1_{X/S})_x
  \]
  is bijective.
  \item[c')] $h$ is smooth at the point $x$, and the canonical homomorphism
  \[
  ((\varOmega^1_{Y/S})_y\otimes_{\sh{O}_{Y,y}}\boldsymbol{k}(y))\otimes_{\mathbf{k}(y)}\boldsymbol{k}(x)\to(\varOmega^1_{X/S})_x\otimes_{\sh{O}_{X,x}}\boldsymbol{k}(x)
  \]
  is bijective.
\end{enumerate}
Suppose moreover that the homomorphism $\boldsymbol{k}(y)\to\boldsymbol{k}(x)$ is bijective. Then the preceding conditions are also equivalent to the following:
\begin{enumerate}
  \item[d)] $h$ is smooth at the point $x$, and the canonical map $\mathrm{T}_{X/S}(x)\to\mathrm{T}_{Y/S}(y)$ (16.5.12) is bijective.
\end{enumerate}
\end{corollary}

Each of the conditions \emph{a)}, \emph{b)}, \emph{c)} of (17.11.2) is equivalent to the conjunction of the corresponding condition of (17.11.1) and of the fact that $g$ is unramified at the point $x$, taking into account (17.2.2) as regards condition c); whence the equivalence of a), b) and c). The equivalence of c) and c') follows from the equivalence of the corresponding conditions of (17.11.1) and from Nakayama's lemma; the equivalence of c') and d) when $\boldsymbol{k}(x)=\boldsymbol{k}(y)$ is immediate by transposition (16.5.12).

\begin{corollary}[17.11.3]
\label{IV.17.11.3}
Let $h:X\to S$ be a smooth morphism, $s_i$ ($1\leq i\leq r$) sections of $\sh{O}_X$ over $X$ (which are also sections of $h_*(\sh{O}_X)$ over $S$), and $g:X\to S[T_1,\ldots,T_r]=\mathbf{V}_S^r$ the $S$-morphism corresponding to the homomorphism of $\sh{O}_S$-modules $\sh{O}_S^r\to h_*(\sh{O}_X)$ defined by these sections \textup{(\textbf{II}, 1.2.7)}. For $g$ to be smooth \textup{(resp.}\ étale\textup{)} at a point $x\in X$, it is necessary and sufficient that the $(d_{X/S}(s_i))_x$ form part of a basis (resp. constitute a basis) of the $\sh{O}_{X,x}$-module $(\varOmega^1_{X/S})_x$.
\end{corollary}

It suffices to apply (17.11.1) (resp. (17.11.2)), taking for $f$ the structural morphism $S[T_1,\ldots,T_r]\to S$.

\begin{proposition}[17.11.5]
\label{IV.17.11.5}
Let $f:Y\to S$, $h:X\to S$ be two smooth morphisms. For an $S$-morphism $g:X\to Y$ to be an open immersion, it is necessary and sufficient that $g$ be a monomorphism of preschemes and that for every $x\in X$, if we set $y=g(x)$, we have $\dim_y(f)=\dim_x(h)$ (for a generalization of this proposition, see (18.10.5)).
\end{proposition}

The condition is evidently necessary; let us prove that it is sufficient.

By virtue of (17.9.5), we are immediately reduced to the case where $S=\Spec(k)$ is the spectrum of a field, and by virtue of (2.7.1, (x)), we may suppose $k$ algebraically closed. Taking into account (17.9.1), it then suffices to prove that $g$ is étale, and since the set of points where $g$ is étale is open, it suffices to show that $g$ is étale at the \emph{closed} points (or equivalently, the points rational over $k$) of $X$ (10.4.8). So let $x$ be such a point, and set $y=g(x)$, which is also rational over $k$; by hypothesis, the ring $A=\sh{O}_{Y,y}$ is regular with residue field $k$; let $d=\dim(A)$ and let $(t_i)_{1\leq i\leq d}$ be a regular system of parameters for $A$. Set $B=\sh{O}_{X,x}$, $C=B/\mathfrak m_yB$. Since $g$ is a monomorphism, so is the morphism $\Spec(C)\to\Spec(\boldsymbol{k}(y))=\Spec(k)$ deduced from $g$ by base change (\textbf{I}, 3.3.12); but this means that the corresponding homomorphism $u:k\to C$ is \emph{surjective} (hence bijective), because $u$ admits a left inverse $v:C\to k$, and $u\circ v$ and the identity of $C$, composed with $u$, give the same morphism $u:k\to C$. Since by hypothesis $B$ is a regular ring of dimension $d$, the images of the $t_i$ in $B$ form a regular system of parameters for $B$ (\textbf{0}, 17.1.7); condition (17.6.3, \emph{e$''$)}) is therefore satisfied by $g$ at the point $x$ (\textbf{0}, 17.1.1 and Bourbaki, \emph{Alg. comm.}, chap. III, § 2, n\textsuperscript{o} 8, cor. 3 of th. 1), which completes the proof.

\begin{theorem}[17.12.1]
\label{IV.17.12.1}
Let $f:X\to S$, $h:Y\to S$ be two morphisms locally of finite presentation, $j:Y\to X$ an immersion, $y$ a point of $Y$, $x=j(y)$. The following conditions are equivalent:
\begin{enumerate}
  \item[\textup{a)}] $h$ is smooth at the point $y$ and $f$ is smooth at the point $x$.

\begin{remark}[17.12.8]
\label{IV.17.12.8}
An $S$-group prescheme $G$, of finite type (and even finite) over a locally Noetherian prescheme $S$, can be differentially smooth over $S$ without being smooth (or even \emph{flat}) over $S$. Take for example for $S$ the spectrum of the algebra of dual numbers $D=k[T]/T^2k[T]$ over a field $k$, and for $G$ the spectrum of the direct product $D$-algebra $E=D\oplus k$. We define on $G$ a structure of $S$-group scheme by defining a ``diagonal map'' $\delta$, a homomorphism from $E$ to $E\otimes_DE$: if $e'$, $e''$ are the idempotents that are the canonical images of the unit element $1$ of $D$ in the factors $D$ and $k$ of $E$, one verifies without difficulty that such a diagonal map is obtained by taking
\[
\delta(e')=e'\otimes e'+e''\otimes e'',\qquad
\delta(e'')=e'\otimes e''+e''\otimes e'.
\]
The unit section of the group scheme $G$ corresponds to the homomorphism $E\to D$ which is the identity on $D$ and $0$ on $k$; it is an isomorphism of $S$ onto a connected component of $G$, and \emph{a fortiori} $G$ is differentially smooth over $S$ (17.12.6), but it is clear that $G$ is not $S$-flat.
\end{remark}

\begin{corollary}[17.13.6]\label{IV.17.13.6}

Then the following conditions are equivalent:
\begin{enumerate}

  \item[\textup{b)}] The homomorphism
  \[
  \rho\otimes1:\varOmega^1_{X/S}\otimes_{\sh{O}_X}\boldsymbol{k}(t)\to
  (\varOmega^1_{Y/S}\otimes_{\sh{O}_Y}\boldsymbol{k}(t))\oplus
  (\varOmega^1_{Z/S}\otimes_{\sh{O}_Z}\boldsymbol{k}(t))
  \]
  where $\rho$ is given by \textup{(17.13.5.4)}, is injective.
  \item[\textup{c)}] The morphism $T=Y\times_SZ\to X\times_SX$ is transversal to the diagonal of $X\times_SX$ at the point $t$.
\end{enumerate}

\begin{enumerate}
  \item[b')] The homomorphism
  \[
  \label{IV.17.13.6.1}
  \tag{17.13.6.1}
  \mathrm{T}_y(f)-\mathrm{T}_z(g):\mathrm{T}_{Y/S}(y)\oplus\mathrm{T}_{Z/S}(z)\to\mathrm{T}_{X/S}(x)
  \]
  \textup{(cf. (16.5.12.5))} is surjective.
\end{enumerate}
Moreover, when conditions a) and b) are satisfied at $t$, they are satisfied in a neighbourhood of $t$ in $T$, and, on restricting $T$ to such a neighbourhood, the sequence
\[
\label{IV.17.13.6.2}
\tag{17.13.6.2}
0\to\varOmega^1_{X/S}\otimes_{\sh{O}_X}\sh{O}_T\xrightarrow{\rho}

\]

\begin{proposition}[17.13.8]

  \rho\otimes1:\varOmega^1_{X/S}\otimes_{\sh{O}_X}\boldsymbol{k}(x)\to
  (\varOmega^1_{Y/S}\otimes_{\sh{O}_Y}\boldsymbol{k}(x))\oplus
  (\varOmega^1_{Z/S}\otimes_{\sh{O}_Z}\boldsymbol{k}(x))

\subsection{Local and infinitesimal characterizations of smooth morphisms, unramified morphisms and étale morphisms.}

\begin{proposition}[17.14.1]
\label{IV.17.14.1}
Let $f:X\to Y$ be a morphism locally of finite presentation, $x$ a point of $X$, $y=f(x)$. For $f$ to be smooth (resp. unramified, resp. étale) at the point $x$, it is necessary and sufficient that the following condition be satisfied:

For every local scheme $Y'=\operatorname{Spec}(A')$ with closed point $y'$, every morphism $h:Y'\to Y$ such that $h(y')=y$, every closed subscheme $Y'_0=\operatorname{Spec}(A'/\mathfrak{J}')$ of $Y'$, where the ideal $\mathfrak{J}'$ is of square zero, and every $Y$-morphism $g_0:Y'_0\to X$ such that $g_0(y')=x$, there exists at least one (resp. at most one, resp. one and only one) $Y$-morphism $g:Y'\to X$ whose restriction to $Y'_0$ is $g_0$.
\end{proposition}

Taking into account the definitions (17.1.1 and 17.3.1), we have to show that the condition of the statement is \emph{sufficient} for $f$ to be smooth (resp. unramified) at the point $x$.

(i) \emph{Case of smooth morphisms.} We may restrict to the case where $Y=\operatorname{Spec}(A)$ and $X=\operatorname{Spec}(C)$ are affine, with $C=B/\mathfrak{K}$, where $B=A[T_1,\ldots,T_n]$ is a polynomial $A$-algebra and $\mathfrak{K}$ an ideal of finite type of $B$. To prove that $f$ is smooth at the point $x$, it suffices to establish that $\sh{O}_{X,x}$ is a formally smooth $\sh{O}_{Y,y}$-algebra for the discrete topologies (17.5.1). Now we have $\sh{O}_{X,x}=B_{\mathfrak q}/\mathfrak{K}B_{\mathfrak q}$, where $\mathfrak q$ is a prime ideal of $B$, and if $\mathfrak p$ is the inverse image of $\mathfrak q$ in $A$, $B_{\mathfrak q}$ is a formally smooth $A_{\mathfrak p}$-algebra for the discrete topologies (\textbf{0}, 19.3.2 and 19.3.5). By application of the Jacobian criterion (\textbf{0}, 22.6.1 and 20.5.12) it therefore suffices to see that $B_{\mathfrak q}/\mathfrak{K}^2B_{\mathfrak q}$ is an $A_{\mathfrak p}$-trivial extension of $B_{\mathfrak q}/\mathfrak{K}B_{\mathfrak q}$ by $\mathfrak{K}B_{\mathfrak q}/\mathfrak{K}^2B_{\mathfrak q}$. But this follows precisely from the hypothesis applied to $A'=B_{\mathfrak q}/\mathfrak{K}^2B_{\mathfrak q}$ and $\mathfrak{J}'=\mathfrak{K}B_{\mathfrak q}/\mathfrak{K}^2B_{\mathfrak q}$ and to the morphism $g_0$ corresponding to the identity automorphism of $A'/\mathfrak{J}'=\sh{O}_{X,x}$.

(ii) \emph{Case of unramified morphisms.} Here set $A=\sh{O}_{Y,y}$, $B=\sh{O}_{X,x}$, and note that by virtue of (17.4.1, \emph{c)}), it suffices to show that $\Omega^1_{B/A}=(\varOmega^1_{X/Y})_x=0$. Now, with the notation of (\textbf{0}, 20.4.1), the ring $C=\mathrm{P}^1_{B/A}=(B\otimes_A B)/\mathfrak{J}^2$ is local, since the same holds for $C/(\mathfrak{J}/\mathfrak{J}^2)$, which is isomorphic to $B$. The hypothesis applied to $A'=C$ and $\mathfrak{J}'=\mathfrak{J}/\mathfrak{J}^2$ shows by definition that the map $d_{B/A}:B\to\Omega^1_{B/A}$ is zero (\textbf{0}, 20.4.6), hence that $\Omega^1_{B/A}=0$ by (\textbf{0}, 20.4.7).

The fact that $X$ is smooth over $k$ at the point $x$ implies the existence of an open neighbourhood $U$ of $x$ that is smooth over $k$; hence $U\otimes_k k'$ is regular for every extension $k'$ of $k$ (17.15.2), which proves that a) implies e). Conversely, e) implies a), since then $U\otimes_k k'$ is smooth over $k'$ (17.15.2), hence $U$ is smooth over $k$ (17.7.1, (ii)).

We have already proved that a) implies c) (17.10.2); c) implies d) trivially, and the fact that d) implies a) follows from (\textbf{0}, 20.4.7) and (17.15.3, \emph{c)}).

Finally, we have already seen that a) implies b) (17.12.4). Conversely, to prove that b) implies a), we may restrict to the case where $x$ is rational over $k$, by considering, as in the proof of (17.15.3), a point $x'$ of $X'=X\otimes_k k'$ above $x$ such that $\boldsymbol{k}(x')=k'=\boldsymbol{k}(x)$, and using again (17.7.1, (ii)) and the fact that if $X$ is differentially smooth at the point $x$, then $X'$ is differentially smooth at the point $x'$ (16.10.4). Supposing therefore that $x$ is rational over $k$, the fact that $f$ is smooth at $x$ then follows from hypothesis b) and from (17.12.5) applied to the $k$-section $u:\operatorname{Spec}(k)\to X$ such that $u(\operatorname{Spec}(k))=\{x\}$.

\begin{corollary}[17.15.6]
\label{IV.17.15.6}
Let $X$ be a prescheme of finite type over a field $k$. For $X$ to be smooth over $k$, it is necessary and sufficient that the $\sh{O}_X$-module $\varOmega^1_{X/k}$ be locally free, and that the local rings at the maximal points of $X$ be fields that are separable extensions of $k$ (this last condition being automatically satisfied if $k$ is a perfect field and $X$ a reduced prescheme).
\end{corollary}

The conditions are necessary, for if $X$ is smooth over $k$, it follows from (17.10.2) that $\varOmega^1_{X/k}$ is locally free; on the other hand, $X$ is regular, hence \emph{a fortiori} reduced, so at every maximal point $x$ of $X$, $\sh{O}_{X,x}$ is a field, which must be a formally smooth $k$-algebra for the discrete topologies (17.5.1), hence a separable extension of $k$ (\textbf{0}, 19.6.1).

The conditions are sufficient. Indeed, we may restrict to the case where $X$ is connected, so that $\varOmega^1_{X/k}$ is locally free of constant rank $n$. For every maximal point $x$ of $X$, we then have $\operatorname{rg}_{\mathbf{k}(x)}\Omega^1_{\mathbf{k}(x)/k}=n$ since $\sh{O}_{X,x}=k(x)$; since by hypothesis $\boldsymbol{k}(x)$ is a separable extension of $k$, we have $\Upsilon_{\mathbf{k}(x)/k}=0$ (\textbf{0}, 20.6.3), hence $\deg.\operatorname{tr}_k \boldsymbol{k}(x)=n$ by Cartier's equality (\textbf{0}, 21.7.1). All the irreducible components of $X$ therefore have the same dimension $n$ (5.2.1), and we conclude that $X$ is smooth over $k$ at every point by virtue of the fact that c) implies a) in (17.15.5).

\begin{corollary}[17.15.7]
\label{IV.17.15.7}
If $k$ is of characteristic 0, then, for $X$ to be smooth over $k$ at the point $x$, it is necessary and sufficient that $(\varOmega^1_{X/k})_x$ be a free $\sh{O}_{X,x}$-module.
\end{corollary}

This follows from (16.12.2) and from the equivalence of b) and a) in (17.15.5).

\begin{remarks}[17.15.10]
\label{IV.17.15.10}
\begin{enumerate}
  \item[(i)] For a prescheme $X$ of finite type over $k$ to be smooth over $k$, it does not suffice that $\varOmega^1_{X/k}$ be locally free, as is shown by the example where $X=\operatorname{Spec}(K)$, $K$ being a finite extension of $k$ that is not separable.
  \item[(ii)] When $k$ is not perfect, it can happen that $X$ is smooth over $k$ without $\boldsymbol{k}(x)$ being separable over $k$. We obtain an example by taking $X=\operatorname{Spec}(k[T])$ and for $x$ the point corresponding to the principal prime ideal $(T^p\setmin\lambda)$ ($p>0$ the characteristic of $k$, $\lambda\in k\setmin k^p$).
  \item[(iii)] However, if $X$ is a prescheme smooth over $k$, the set of closed points of $X$ such that $\boldsymbol{k}(x)$ is separable (and finite) over $k$ is dense in $X$. Indeed, let $f:X\to\operatorname{Spec}(k)$ be the structural morphism; for every $x_0\in X$, there are an open neighbourhood $U$ of $x_0$ in $X$ and a factorization of $f|U:U\xrightarrow{g}\operatorname{Spec}(k[T_1,\ldots,T_n])\xrightarrow{h}\operatorname{Spec}(k)$ where $g$ is étale (17.11.4). Since we may restrict to the case where $k$ is not perfect, hence infinite, the set of points of the open subset $g(U)$ that are rational over $k$ is non-empty; if $y$ is such a point and $x\in U$ a point above $y$, then $x$ is closed in $X$ and $\boldsymbol{k}(x)$ is separable over $k(y)=k$ (17.6.2).
\end{enumerate}
\end{remarks}

\begin{proposition}[17.15.11]

  \item[\textup{b)}] $X$ is unramified over $k$.

\end{enumerate}

\begin{proposition}[17.15.12]
\label{IV.17.15.12}
Let $k$ be a field and $X$ a prescheme locally of finite type over $k$. For there to exist an everywhere dense open subset $U$ of $X$ that is smooth over $k$, it is necessary and sufficient that for every maximal point $x$ of $X$, $X$ be reduced at this point and $\boldsymbol{k}(x)$ be a separable extension of $k$. For there to exist an everywhere dense open subset $U$ of $X$ such that $U_{\mathrm{red}}$ is smooth over $k$, it is necessary and sufficient that for every maximal point $x$ of $X$, $\boldsymbol{k}(x)$ be a separable extension of $k$.
\end{proposition}

The second assertion evidently follows from the first. To say that there exists an everywhere dense open subset $U$ smooth over $k$ means that $X$ is smooth over $k$ at each of its maximal points $x$. For this it is necessary that $\sh{O}_{X,x}$ be regular (17.15.1) and \emph{a fortiori} reduced, hence a field, equal to $\boldsymbol{k}(x)$; moreover (17.15.1), $\boldsymbol{k}(x)\otimes_k k'$ must be regular for every radicial extension $k'$ of $k$, so $\boldsymbol{k}(x)$ must be a separable extension of $k$ (4.3.5); the converse is immediate, by virtue of (17.15.1).
