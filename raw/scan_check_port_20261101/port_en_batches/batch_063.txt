\begin{env}[17.4.5]\label{IV.17.4.5}
\EGAReaderTarget{ega4.c04.statement.17-4-5}{IV.17.4.5}
Given an $S$-prescheme $Y$ and two $S$-morphisms $f:X\to Y$, $g:X\to Y$, we canonically deduce from them an $S$-morphism $(f,g)_S:X\to Y\times_S Y$. We will call the \emph{prescheme of coincidences} of $f$ and $g$ the inverse image under $(f,g)_S$ of the diagonal $\Delta_{Y|S}$; it is therefore a sub-prescheme of $X$, which is \emph{closed} when $Y$ is an $S$-scheme (\textbf{I}, 5.4.1).
\end{env}

\begin{proposition}[17.4.6]
\label{IV.17.4.6}
Let $h:Y\to S$ be an unramified morphism, and let $f:X\to Y$, $g:X\to Y$ be two $S$-morphisms. Then the prescheme $C$ of coincidences of $f$ and $g$ is a sub-prescheme induced on an open subset of $X$; if moreover $Y$ is an $S$-scheme \textup{(\textbf{I}, 5.4.1)}, $C$ is a closed sub-prescheme of $X$.
\end{proposition}

Indeed, since $\Delta_h:Y\to Y\times_S Y$ is an open immersion (17.4.2), the inverse image under $(f,g)_S$ of $\Delta_{Y|S}$ is a sub-prescheme induced on an open subset of $X$ (\textbf{I}, 4.4.1). The last assertion follows from (17.4.5).

\begin{corollary}[17.4.7]
\label{IV.17.4.7}
Under the hypotheses of \textup{(17.4.6)}, let $x$ be a point of $X$ such that the two composite morphisms $\Spec(\boldsymbol{k}(x))\to X\xrightarrow{f}Y$ and $\Spec(\boldsymbol{k}(x))\to X\xrightarrow{g}Y$ are equal. Then there exists an open neighbourhood $U$ of $x$ such that $f|U=g|U$. If moreover $Y$ is an $S$-scheme, there exists an open and closed neighbourhood $X'$ of $x$ in $X$ such that $f|X'=g|X'$. If finally we suppose in addition that $X$ is connected, we have $f=g$.
\end{corollary}

This follows from (17.4.6) and from (\textbf{I}, 5.3.17).

\begin{corollary}[17.4.8]
\label{IV.17.4.8}
Under the hypotheses of \textup{(17.4.6)}, suppose that the structural morphism $\varphi=h\circ f=h\circ g$ from $X$ to $S$ is closed. \begin{proposition}[17.4.9]
\label{IV.17.4.9}
Let $Y$ be a connected prescheme, and $f:X\to Y$ an unramified and separated morphism. Then every $Y$-section $g$ of $X$ is an isomorphism of $Y$ onto an open connected component of $X$, and the map $g\leadsto g(Y)$ is a bijection of $\Gamma(X/Y)$ onto the set of connected components $Z$ of $X$ (necessarily open in $X$) such that the restriction of $f$ to $Z$ is an isomorphism of $Z$ onto $Y$. In particular, if $g'$ and $g''$ are two $Y$-sections of $X$ such that $g'(y)=g''(y)$ for some $y\in Y$, we have $g'=g''$.
\end{proposition}

Indeed, it follows from (17.4.1, \emph{b'')}) that a $Y$-section $s$ of $X$ is an open immersion, and as $X$ is a $Y$-scheme, $s$ is also a closed immersion (\textbf{I}, 5.4.7); it follows that $s$ is an isomorphism of $Y$ onto a sub-prescheme of $X$ induced on an open and closed subset of $X$, and as $s(Y)$ is connected, it is necessarily a connected component of $X$. The rest of the proposition is immediate.

\begin{theorem}[17.5.1]
\label{IV.17.5.1}
Let $f:X\to Y$ be a morphism locally of finite presentation, $x$ a point of $X$, and $y=f(x)$. The following conditions are equivalent:
\begin{enumerate}
  \item[\textup{a)}] $f$ is smooth at the point $x$.
  \item[\textup{b)}] $f$ is flat at the point $x$ and the $\boldsymbol{k}(y)$-prescheme $f^{-1}(y)$ is smooth over $\boldsymbol{k}(y)$ at the point $x$.
  \item[b')] $f$ is regular at the point $x$ \textup{(6.8.1)}.
  \item[\textup{c)}] The ring $\sh{O}_{X,x}$ is a formally smooth $\sh{O}_{Y,y}$-algebra for the discrete topologies.
\end{enumerate}
\end{theorem}

We may restrict to the case where $Y=\Spec(A)$ and $X=\Spec(C)$, where $C=B/\mathfrak{J}$, $B=A[T_1,\ldots,T_n]$ being a polynomial algebra and $\mathfrak{J}$ an ideal of $B$ of finite type. The equivalence of a) and c) then follows from the equivalence of a) and c) in (\textbf{0}, 22.6.4). On the other hand, applying this result to the morphism locally of finite type $f^{-1}(y)\to\Spec(\boldsymbol{k}(y))$, we see that the equivalence of b) and b') follows from the equivalence of a) and b) in (6.8.6). It therefore remains to prove the equivalence of a) and b).

Let us first show that a) implies b); denote by $\mathfrak p$ the prime ideal $\mathfrak j_x$ in $C$, and by $\mathfrak r$ the prime ideal $\mathfrak j_y$ in $A$; we have $\mathfrak p=\mathfrak q/\mathfrak J$, where $\mathfrak q$ is a prime ideal of $B$, and $\mathfrak r$ is the inverse image of $\mathfrak q$ in $A$. Hypothesis a) implies first that $f^{-1}(y)$ is smooth over $\boldsymbol{k}(y)$ at the point $x$, by (17.3.3), and we must show moreover that $C_{\mathfrak p}=\sh{O}_{X,x}$ is a \emph{flat} $A_{\mathfrak r}$-module. As $C_{\mathfrak p}$ is a formally smooth $A_{\mathfrak r}$-algebra and $B$ is a formally smooth $A$-algebra (for the discrete topologies), the Jacobian criterion (\textbf{0}, 22.6.4), together with (\textbf{0}, 19.1.12), implies that there exist in $\mathfrak J$ a system of $r$ polynomials $u_i$ ($1\leq i\leq r$) and $r$ indices $j_i$ ($1\leq i\leq r$) such that the images of the $u_i$ in $\mathfrak J_{\mathfrak q}/\mathfrak J_{\mathfrak q}^2$ generate this $B_{\mathfrak q}$-module and such that
\[
\label{IV.17.5.1.1}
\det(\partial u_i/\partial T_{j_k})\notin\mathfrak q.
\tag{17.5.1.1}
\]

Let us now note that if $g:\Spec(B)\to\Spec(A)$ is the structural morphism, the fibre $g^{-1}(y)$ is the spectrum of the regular ring $\boldsymbol{k}(y)[T_1,\ldots,T_n]$ (\textbf{0}, 17.3.7), so the Noetherian local ring $B_{\mathfrak q}/\mathfrak rB_{\mathfrak q}$ at a point of this fibre is regular. Now, condition (17.5.1.1) implies that the canonical images $v_i$ of the $u_i$ in the maximal ideal $\mathfrak m$ of $B_{\mathfrak q}/\mathfrak rB_{\mathfrak q}$ are \emph{linearly independent} mod. $\mathfrak m^2$: otherwise, indeed, there would exist polynomials $w_i\in B$ ($1\leq i\leq r$), not all belonging to $\mathfrak q$, such that $\sum_{i=1}^r w_i u_i\in\mathfrak q^2$. Differentiating with respect to the $T_{j_k}$, we would conclude that $\sum_{i=1}^r w_i(\partial u_i/\partial T_{j_k})\in\mathfrak q$ for $1\leq k\leq r$, which would contradict (17.5.1.1) since $\mathfrak q$ is prime. We therefore conclude from (\textbf{0}, 17.1.7) that $(v_i)$ is a \emph{regular sequence} in $B_{\mathfrak q}/\mathfrak rB_{\mathfrak q}$. But as the morphism $g$ is locally of finite presentation and $B$ is a flat $A$-module, it follows from (11.3.8) that the canonical images $u'_i$ of the $u_i$ in $B_{\mathfrak q}$ also form a \emph{regular sequence} and that $B_{\mathfrak q}/(\sum_i u'_iB_{\mathfrak q})$ is a \emph{flat} $A_{\mathfrak r}$-module. As the images of the $u'_i$ in $\mathfrak J_{\mathfrak q}/\mathfrak J_{\mathfrak q}^2$ generate this $B_{\mathfrak q}$-module, it follows from Nakayama's lemma that $\sum_i u'_iB_{\mathfrak q}=\mathfrak J_{\mathfrak q}$, and $C_{\mathfrak p}=B_{\mathfrak q}/\mathfrak J_{\mathfrak q}$ is therefore indeed a flat $A_{\mathfrak r}$-module.

Finally, let us prove that b) implies a). With the same notation, the hypothesis that $B_{\mathfrak q}/\mathfrak J_{\mathfrak q}$ is a flat $A_{\mathfrak r}$-module implies that the canonical homomorphism $\mathfrak J_{\mathfrak q}/\mathfrak r\mathfrak J_{\mathfrak q}\to B_{\mathfrak q}/\mathfrak rB_{\mathfrak q}$ is \emph{injective} ($\mathbf{0}_{\mathrm I}$, 6.1.2), so that $\mathfrak J_{\mathfrak q}/\mathfrak r\mathfrak J_{\mathfrak q}$ is identified with an ideal of $B_{\mathfrak q}/\mathfrak rB_{\mathfrak q}$. As $B_{\mathfrak q}/\mathfrak rB_{\mathfrak q}$ is a formally smooth $A_{\mathfrak r}$-algebra for the discrete topologies, we can apply the Jacobian criterion (\textbf{0}, 22.6.4) to $C_{\mathfrak p}=(B_{\mathfrak q}/\mathfrak rB_{\mathfrak q})/(\mathfrak J_{\mathfrak q}/\mathfrak r\mathfrak J_{\mathfrak q})$; together with (\textbf{0}, 19.1.12), this criterion proves, by virtue of hypothesis b), the existence of $r$ polynomials $v_i\in\boldsymbol{k}(y)[T_1,\ldots,T_n]$ such that their images in $(\mathfrak J_{\mathfrak q}/\mathfrak r\mathfrak J_{\mathfrak q})/(\mathfrak J_{\mathfrak q}/\mathfrak r\mathfrak J_{\mathfrak q})^2$ generate this $(B_{\mathfrak q}/\mathfrak rB_{\mathfrak q})$-module and such that
\[
\label{IV.17.5.1.2}
\det(\partial v_i/\partial T_{j_k})\notin\mathfrak qB_{\mathfrak q}/\mathfrak rB_{\mathfrak q}.
\tag{17.5.1.2}
\]
If, for every $i$, we then denote by $u_i$ an element of $\mathfrak J$ whose canonical image is $v_i$, it follows from (17.5.1.2) that the $u_i$ satisfy condition (17.5.1.1); on the other hand, by virtue of Nakayama's lemma, the images $u'_i$ of the $u_i$ in $\mathfrak J_{\mathfrak q}$ generate this $B_{\mathfrak q}$-module. The Jacobian criterion (\textbf{0}, 22.6.4), together with (\textbf{0}, 19.1.12), then proves that $C_{\mathfrak p}=B_{\mathfrak q}/\mathfrak J_{\mathfrak q}$ is a formally smooth $A_{\mathfrak r}$-algebra for the discrete topologies. Q.E.D.

\begin{corollary}[17.5.2]
\label{IV.17.5.2}
Let $f:X\to Y$ be a morphism locally of finite presentation. For $f$ to be smooth (in the sense of (17.3.1)), it is necessary and sufficient that $f$ be regular \textup{(6.8.1)}, in other words that $f$ be flat and that for every $y\in Y$, $f^{-1}(y)$ be a geometrically regular $\boldsymbol{k}(y)$-prescheme \textup{(6.7.6)}.
\end{corollary}

We have thus established the equivalence of the two definitions of ``smooth morphism'' given in (6.8.1) and (17.3.1).

\begin{proposition}[17.5.5]
\label{IV.17.5.5}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, and $y=f(x)$. Suppose that $Y$ is reduced at the point $y$. Then, for $f$ to be smooth at the point $x$, it is necessary and sufficient that $f$ be universally open in a neighbourhood of $x$ in $f^{-1}(y)$ and that $f^{-1}(y)$ be a $\boldsymbol{k}(y)$-prescheme geometrically regular at the point $x$.
\end{proposition}

Taking into account (17.5.1), everything comes down to seeing that, if $f^{-1}(y)$ is a $\boldsymbol{k}(y)$-prescheme geometrically regular at the point $x$, it is equivalent to say that $f$ is flat at the point $x$ or that it is universally open in a neighbourhood of $x$ in $f^{-1}(y)$. Now, if $f$ is flat at the point $x$, it is flat in a neighbourhood of $x$ in $X$ (11.1.1), and consequently it is universally open in this neighbourhood (2.4.6). Conversely, the hypothesis that $f$ is universally open in a neighbourhood of $x$ in $f^{-1}(y)$ and that $f^{-1}(y)$ is a $\boldsymbol{k}(y)$-prescheme geometrically regular at the point $x$ implies that $f$ is flat at the point $x$, since $\sh{O}_{Y,y}$ is reduced (15.2.2).

\begin{corollary}[17.5.6]
\label{IV.17.5.6}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, and $y=f(x)$. Suppose that $Y$ is reduced and geometrically unibranch (6.15.1) at the point $y$. Then, for $f$ to be smooth at the point $x$, it is necessary and sufficient that $f$ be equidimensional at the point $x$ and that $f^{-1}(y)$ be a $\boldsymbol{k}(y)$-prescheme geometrically regular at the point $x$.
\end{corollary}

Noting that the set of points where $f$ is equidimensional is open (13.3.2), we see that the corollary follows from (17.5.5) and from Chevalley's criterion (14.4.4).

\begin{corollary}[17.6.2]
\label{IV.17.6.2}
Let $f:X\to Y$ be a morphism locally of finite presentation. The following conditions are equivalent:
\begin{enumerate}
  \item[\textup{a)}] $f$ is étale.
  \item[a')] $f$ is smooth and unramified.
  \item[\textup{b)}] $f$ is smooth and locally quasi-finite \textup{(\textbf{Err}$_{\mathrm{III}}$, 20)}.
  \item[\textup{c)}] $f$ is flat and unramified.
  \item[c')] $f$ is flat, and every fibre $f^{-1}(y)$ is a $\boldsymbol{k}(y)$-scheme which is a sum of spectra of fields that are finite and separable extensions of $\boldsymbol{k}(y)$.
  \item[c'')] $f$ is flat, and for every $y\in Y$ and every algebraically closed extension $k'$ of $\boldsymbol{k}(y)$, the ``geometric fibre'' $f^{-1}(y)\otimes_{\mathbf{k}(y)}k'$ is a sum of spectra of fields isomorphic to $k'$.
\end{enumerate}
\end{corollary}

The only point that remains to be proved is the equivalence of c') and c''). It is clear that c') implies c'') by base change (17.3.3). On the other hand, as the projection morphism $f^{-1}(y)\otimes_{\mathbf{k}(y)}k'\to f^{-1}(y)$ is open (2.4.10), hypothesis c'') implies that the space $f^{-1}(y)$ is discrete; hence, for every point $x\in X_y=f^{-1}(y)$, the local ring $\sh{O}_{X_y,x}=A$ is a finite $\boldsymbol{k}(y)$-algebra, hence an Artinian local ring; moreover it follows from c'') that $\Spec(A\otimes_{\mathbf{k}(y)}k')$ is a sum of spectra of fields isomorphic to $k'$, which is possible only if $A$ is a field which is a finite separable extension of $\boldsymbol{k}(y)$ (4.6.1).

\begin{proposition}[17.6.3]
\label{IV.17.6.3}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, and $y=f(x)$. Set $A=\sh{O}_{Y,y}$, $B=\sh{O}_{X,x}$, which are Noetherian local rings, and let $k$ be the residue field of $A$. Then the equivalent conditions a) to d) of (17.6.1) are also equivalent to each of the following:
\begin{enumerate}
  \item[\textup{e)}] $\widehat B$ is a formally étale $\widehat A$-algebra for the adic topologies.
  \item[e')] $\widehat B$ is a free $\widehat A$-module and $\widehat B\otimes_{\widehat A}k$ is a field which is a finite and separable extension of $k$ (which implies that $\widehat B$ is a finite $\widehat A$-algebra).
\end{enumerate}
If moreover $\boldsymbol{k}(x)=\boldsymbol{k}(y)$, or if $k$ is separably closed, these conditions are also equivalent to:
\begin{enumerate}
  \item[e'')] The canonical homomorphism $\widehat A\to\widehat B$ is bijective.
\end{enumerate}
\end{proposition}

The equivalence of e) and each of the conditions of (17.6.1) follows immediately from (17.4.4, f')) and (17.5.3, d')). The fact that e) implies e') follows from (17.4.4, f)) and from (\textbf{0}, 19.7.1), taking into account that $\widehat B$ is then a finite $\widehat A$-algebra (17.4.4), so that it amounts to the same to say that $\widehat B$ is a flat $\widehat A$-module or a free $\widehat A$-module ($\mathbf{0}_{\mathrm{III}}$, 10.1.3). Conversely, the fact that e') implies e) follows from (17.4.4) and from (\textbf{0}, 19.7.1). Finally, \emph{e')} implies that the homomorphism $\widehat A\to\widehat B$ is injective, and if $\boldsymbol{k}(x)=\boldsymbol{k}(y)$ or if $k$ is separably closed, this homomorphism is surjective by (17.4.4). The converse is immediate.

\begin{proposition}[17.7.1]
\label{IV.17.7.1}
Let $f:X\to Y$ be a morphism locally of finite presentation, $g:Y'\to Y$ a morphism, $X'=X\times_Y Y'$, and $f'=f_{(Y')}:X'\to Y'$ and $g':X'\to X$ the canonical projections. Let $x'$ be a point of $X'$, and set $x=g'(x')$, $y'=f'(x')$.
\begin{enumerate}
  \item[\textup{(i)}] If $f'$ is unramified at the point $x'$, then $f$ is unramified at the point $x$.
  \item[\textup{(ii)}] Suppose moreover that $g$ is flat at the point $y'$. Then, if $f'$ is smooth (resp. étale) at the point $x'$, $f$ is smooth (resp. étale) at the point $x$.
\end{enumerate}
\end{proposition}

Set $y=f(x)=g(y')$, so that we have $f'^{-1}(y')=f^{-1}(y)\otimes_{\mathbf{k}(y)}k(y')$; note that $f'$ is locally of finite presentation.

(i) As the property, for a morphism (locally of finite presentation), of being unramified at a point involves only the fibre of the morphism at this point (17.4.1, $d$)), we may restrict to the case where $Y$ and $Y'$ are spectra of fields. But then it amounts to the same to say that $f$ (resp. $f'$) is unramified at $x$ (resp. $x'$) or that it is étale at this point (17.6.1, $c$)), so (i) is a consequence of (ii).

(ii) As $f'$ is flat at the point $x'$ (17.5.1), the hypothesis that $g$ is flat at the point $y'$ implies that $f$ is flat at the point $x$, because the projection $g':X'\to X$ is a morphism flat at the point $x'$, and $f\circ g'=g\circ f'$ is a morphism flat at the point $X'$, whence the conclusion (2.2.11, (iv)). The fact that $f$ is smooth (resp. étale) at the point $x$ then involves only the fibre $f^{-1}(y)$ (17.5.1 and 17.6.1), and we are therefore again reduced to the case where $Y=\Spec(k)$ and $Y'=\Spec(k')$ are spectra of fields. To say that $f'$ is smooth at the point $x'$ then means (17.5.1) that $X'$ is a $k'$-prescheme geometrically regular at the point $x'$, and this implies (6.7.8) that $X$ is a $k$-prescheme geometrically regular at the point $x$, hence that $f$ is smooth at the point $x$. Suppose moreover that $f'$ is étale at the point $x'$, so that $x'$ is isolated in $f'^{-1}(y')$ (17.6.1); as the projection $f'^{-1}(y')\to f^{-1}(y)$ is an open morphism (2.4.10), $x$ is isolated in $f^{-1}(y)$; as we already know that $f$ is smooth at the point $x$, it is étale at this point (17.6.1).

\label{IV.17.7.2}
\begin{enumerate}
  \item[\textup{(i)}] With the notation of (17.7.1), let $U$ (resp. $U'$) be the set of points of $X$ (resp. $X'$) where $f$ (resp. $f'$) is unramified; then we have $U'=g'^{-1}(U)$.
  \item[\textup{(ii)}] Suppose moreover that $g$ is flat, and let $V$ (resp. $V'$) be the set of points of $X$ where $f$ (resp. $f'$) is smooth; then $V'=g'^{-1}(V)$.

\begin{proposition}[17.7.10]
\label{IV.17.7.10}
Let $f:Y\to S$, $h:X\to S$ be two morphisms locally of finite presentation, $g:X\to Y$ an $S$-morphism, $x$ a point of $X$, and $y=g(x)$. Suppose that $h$ is flat at the point $x$ and that $f$ is unramified at the point $y$. Then $f$ is étale at the point $y$ and $g$ is flat at the point $x$.
\end{proposition}

(We will see later (18.4.9) that we can in fact dispense with the hypothesis that $h$ is locally of finite presentation).

The question being local on $S$, $X$ and $Y$, we may suppose that $S$, $X$ and $Y$ are affine, $f,g,h$ morphisms of finite presentation, $h$ flat and $f$ unramified. Taking into account (11.2.7) and (17.7.9), we may suppose moreover that $S$, $X$ and $Y$ are Noetherian; finally, we may restrict to the case where $S=\Spec(\sh{O}_{S,s})$ (where $s=f(y)=h(x)$). The ring $A=\sh{O}_{S,s}$ being a Noetherian local ring, there exist a complete Noetherian local ring $B$ with algebraically closed residue field and a local homomorphism $A\to B$ making $B$ a faithfully flat $A$-module ($\mathbf{0}_{\mathrm{III}}$, 10.3.1). Replacing $X$ and $Y$ by $X\times_S\Spec(B)$ and $Y\times_S\Spec(B)$, we conclude by (2.5.1) and (17.7.1) that we may restrict to the case where $A$ is complete and has an algebraically closed residue field. The hypothesis that $f$ is unramified at the point $y$ then implies (17.4.4) that $\sh{O}_{Y,y}$ is a finite $\sh{O}_{S,s}$-algebra and a complete Noetherian local ring ($\mathbf{0}_{\mathrm I}$, 7.4.2), and as the residue field of $\sh{O}_{S,s}$ is algebraically closed, the homomorphism $\sh{O}_{S,s}\to\sh{O}_{Y,y}$ is surjective (17.4.4). But, on the other hand, the hypothesis that $h$ is flat at the point $x$ implies that the composite homomorphism $\sh{O}_{S,s}\to\sh{O}_{Y,y}\to\sh{O}_{X,x}$ is injective ($\mathbf{0}_{\mathrm I}$, 6.5.1), hence the homomorphism $\sh{O}_{S,s}\to\sh{O}_{Y,y}$ is bijective, which shows that $f$ is étale at the point $y$ (17.6.3); moreover, $\sh{O}_{X,x}$ is a flat $\sh{O}_{Y,y}$-module, hence $g$ is flat at the point $x$.

\subsection{Criteria for smoothness and unramifiedness by fibres.}
\EGAReaderTarget{ega4.c04.subsec.17-8}{subsection:IV.17.8}
\label{IV.17.8}

\begin{proposition}[17.8.1]
\label{IV.17.8.1}
Let $g:Y\to S$, $h:X\to S$ be two morphisms locally of finite presentation. For an $S$-morphism $f:X\to Y$ to be unramified, it is necessary and sufficient that for every $s\in S$, the morphism $f_s:h^{-1}(s)\to g^{-1}(s)$ deduced from $f$ by the base change $\Spec(\boldsymbol{k}(s))\to S$ be unramified.
\end{proposition}

This follows immediately from the fact that, for a morphism locally of finite presentation, being unramified is a property of the \emph{fibres} of this morphism (17.4.1, $d$)), and from the fact that, for every $y\in Y$, we have $f^{-1}(y)=f_s^{-1}(y)$ if $s=g(y)$.

\begin{proposition}[17.8.2]
\label{IV.17.8.2}
Let $g:Y\to S$, $h:X\to S$ be two morphisms locally of finite presentation; suppose moreover that $h$ is flat. For an $S$-morphism $f:X\to Y$ to be smooth \textup{(resp.}\ étale\textup{)}, it is necessary and sufficient that, for every $s\in S$, the morphism $f_s:h^{-1}(s)\to g^{-1}(s)$ deduced from $f$ by the base change $\Spec(\boldsymbol{k}(s))\to S$ be smooth \textup{(resp.}\ étale\textup{)}. When this is the case, the morphism $g$ is flat at the points of $f(X)$.
\end{proposition}

Indeed, we know (11.3.10) that for $f$ to be flat, it is necessary and sufficient that $f_s$ be flat for every $s\in S$, and that $g$ is then flat at the points of $f(X)$. But for a flat morphism locally of finite presentation, being smooth is a property of the \emph{fibres} of this morphism (17.5.1, $b$)), and for every $y\in Y$, we have $f^{-1}(y)=f_s^{-1}(y)$ if $s=g(y)$.

\begin{remark}[17.8.3]
\label{IV.17.8.3}
The preceding proofs show (taking into account (11.3.10)) that if the hypotheses on $g$ and $h$ are the same as above, then, for $f$ to be unramified (resp. smooth, resp. étale) \emph{at a point} $x\in X$, it suffices that, if we set $s=h(x)$, $f_s$ be unramified (resp. smooth, resp. étale) at the point $x$.
\end{remark}

\subsection{Relative dimension of a prescheme smooth over another}
\EGAReaderTarget{ega4.c04.subsec.17-10}{subsection:IV.17.10}

\begin{definition}[17.10.1]\label{IV.17.10.1}
\EGAReaderTarget{ega4.c04.def.17-10-1}{IV.17.10.1}
Let $f:X\to Y$ be a morphism locally of finite type. We call \emph{relative dimension of $f$ at the point $x\in X$} (or \emph{relative dimension of $X$ over $Y$ at the point $x$}), and denote by $\dim_x f$, the nonnegative integer $\dim_x(f^{-1}(f(x)))$.
\end{definition}

To say that $f$ is \emph{quasi-finite} at the point $x$ (\textbf{Err}$_{\mathrm{III}}$, 20) is therefore equivalent to saying that $\dim_x f=0$. We have seen (13.1.3) that the function $x\leadsto\dim_x f$ is upper semi-continuous. Note that, even when the morphism $f$ has property $(S_1)$ (in other words (6.8.1) is flat and such that its fibres have no embedded associated prime cycle), the function $x\leadsto\dim_x f$ is not necessarily continuous, as is shown by the example where $Y=\Spec(k)$, with $k$ a field, and $X=\Spec(k[U,V,W]/\mathfrak p\mathfrak q)$, where $\mathfrak p=(W)$ and $\mathfrak q=(U)+(V-W)$, prime ideals of $k[U,V,W]$ ($X$ thus being the union, in 3-dimensional space, of a plane and a line not parallel to this plane).

To say that a morphism $f:X\to Y$ locally of finite presentation is étale at the point $x\in X$ also means that $f$ is smooth at the point $x$ and that $\dim_x f=0$ (17.6.1).
