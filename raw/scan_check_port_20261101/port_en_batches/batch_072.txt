Finally, when $\mathrm{A}$ is Noetherian, the $f_i$ belong to the radical of $\mathrm{A}$, the sequence $\mathbf{f}$ is $\operatorname{gr}_\mathfrak K^*(\mathrm{A})$-regular, and $\mathrm{M}$ is a finitely generated $\mathrm{A}$-module, the conditions \emph{a)}, \emph{b)}, \emph{c)}, \emph{d)} are equivalent.
\end{theorem}

\begin{proof}
The implication \emph{a)} $\Rightarrow$ \emph{c)} is immediate $(\mathbf{0},15.1.13)$. If $c)$ is satisfied, it results from \sref{IV.19.5.5} that $\psi_\mathrm{M}$ is bijective; moreover, $(\operatorname{gr}_\mathfrak K^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ is a flat $(\mathrm{A}/\mathfrak{L})$-module, and hence so is $\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})$, since $\psi_\mathrm{M}$ is an $(\mathrm{A}/\mathfrak{L})$-isomorphism; thus $c)$ implies $b)$. The fact that $a)$ also implies $d)$ is immediate.

When every quotient of a submodule of $\operatorname{gr}_\mathfrak K^k(\mathrm{M})$ is ideally separated for $\mathfrak{J}$, it results from \sref{IV.19.5.5} that condition $b)$ implies that $\mathbf{f}$ is $(\operatorname{gr}_\mathfrak K^*(\mathrm{M}))$-regular; moreover, $(\operatorname{gr}_\mathfrak K^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ is isomorphic to $\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})$ and hence is flat over $\mathrm{A}/\mathfrak{L}$. Its degree-zero direct summand $\operatorname{gr}_\mathfrak K^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L})$ is therefore flat; this proves that $b)$ implies $c)$. On the other hand, under the same hypothesis, $c)$ implies $a)$ by virtue of $(\mathbf{0},15.1.21)$ (where we can replace the Noetherian hypothesis by the hypothesis that $\mathrm{M}$ is ideally separated for $\mathfrak{J}$, by virtue of $(\mathbf{0}_\mathrm{III},10.2.1)$).

It remains therefore to prove that when $\mathrm{A}$ is Noetherian, $\mathrm{M}$ of finite type, and the $f_i$ belong to the radical of $\mathrm{A}$, $d)$ implies $a)$.

There exists a free $\mathrm{A}$-module $\mathrm{N}$ of \emph{finite type} and an exact sequence $0\to\mathrm{R}\to\mathrm{N}\to\mathrm{M}\to 0$. It suffices to prove that $d)$ implies $b)$, since every quotient of a sub-$\operatorname{gr}_\mathfrak K^k(\mathrm{M})$ is then ideally separated for $\mathfrak{J}$, by virtue of the fact that $\operatorname{gr}_\mathfrak K^k(\mathrm{M})$ is a finitely generated $\mathrm{A}$-module, that $\mathfrak{J}$ is contained in the radical of $\mathrm{A}$, and that $\mathrm{A}$ is Noetherian (Bourbaki, \emph{Alg.~comm.}, chap.~III, \textsection~5, n\textsuperscript{o}~1); in other words, it suffices to show that $\psi_\mathrm{M}$ is \emph{bijective}.

As $\mathrm{N}$ is free and the sequence $\mathbf{f}$ is $\operatorname{gr}_\mathfrak K^*(\mathrm{A})$-regular by hypothesis, it is also $\operatorname{gr}_\mathfrak K^*(\mathrm{N})$-regular. One can therefore apply \sref{IV.19.6.3}, since every quotient module of a sub-$\operatorname{gr}_k(\mathrm{M})$ is separated for the $\mathfrak{J}$-preadic topology, the $f_i$ belonging to the radical of $\mathrm{A}$ by hypothesis. The diagram \sref{IV.19.6.2.1} here reads
\[
\xymatrix{
  & 0 \ar[d] \\
  (\operatorname{gr}_\mathfrak K^*(\mathrm{R},\mathrm{N})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \ar[r]^-{\psi_\mathrm{R}'} \ar[d] & \operatorname{gr}_\mathfrak{L}^*(\mathrm{R},\mathrm{N}) \ar[d] \\
  (\operatorname{gr}_\mathfrak K^*(\mathrm{N})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \ar[r]^-{\psi_\mathrm{N}} \ar[d] & \operatorname{gr}_\mathfrak{L}^*(\mathrm{N}) \ar[d] \\
  (\operatorname{gr}_\mathfrak K^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \ar[r]^-{\psi_\mathrm{M}} \ar[d] & \operatorname{gr}_\mathfrak{L}^*(\mathrm{M}) \ar[d] \\
  0 & 0
}
\tag{19.7.1.1}
\label{IV.19.7.1.1}
\]
where $\operatorname{gr}_\mathfrak K^*(\mathrm{R},\mathrm{N})$ (resp. $\operatorname{gr}_\mathfrak{L}^*(\mathrm{R},\mathrm{N})$) is the graded associated to $\mathrm{R}$ for the filtration $\mathrm{R}\cap\mathfrak K^m\mathrm{N}$ (resp. $\mathrm{R}\cap\mathfrak{L}^m\mathrm{N}$); recall that in this diagram the columns are exact and that $\psi_\mathrm{N}$ is \emph{bijective}.

Set $\mathrm{B}=\operatorname{gr}_\mathfrak{R}^*\mathrm{A}\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L})$, $\mathrm{P}=\operatorname{gr}_\mathfrak{R}^*(\mathrm{N})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L})$, $\mathrm{Q}=\operatorname{gr}_\mathfrak{L}^*(\mathrm{R},\mathrm{N})$; since $\psi_\mathrm{N}$ is bijective, $\mathrm{P}[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ identifies with $\operatorname{gr}_\mathfrak{L}^*(\mathrm{N})$, and $\mathrm{Q}$ is a sub-$\mathrm{B}[\mathrm{T}_1,\ldots,\mathrm{T}_n]$-module of $\mathrm{P}[\mathrm{T}_1,\ldots,\mathrm{T}_n]$; we will first show that we have

  \end{aligned}

As an $(\mathrm{A}/\mathfrak{L})$-module, it is therefore isomorphic to a submodule of a direct sum of $(\mathrm{A}/\mathfrak{L})$-modules isomorphic to $\mathrm{P}/(\mathrm{Q}\cap\mathrm{P})=(\mathrm{P}+\mathrm{Q})/\mathrm{Q}$.
But $(\mathrm{P}+\mathrm{Q})/\mathrm{Q}$ is an $(\mathrm{A}/\mathfrak{L})$-module isomorphic to a submodule of $\mathrm{P}[\mathrm{T}_1,\ldots,\mathrm{T}_n]/\mathrm{Q}$, which is none other than $\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})$ by the diagram \sref{IV.19.7.1.1}.

\begin{equation}
  \operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{Z})\subset\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})).
  \tag{19.7.1.4}\label{IV.19.7.1.4}
\end{equation}

But each $\operatorname{gr}_\mathfrak{L}^m(\mathrm{M})$ is by hypothesis a \emph{flat} $(\mathrm{A}/\mathfrak{L})$-module of finite type, hence projective since $\mathrm{A}/\mathfrak{L}$ is Noetherian, and we therefore have $\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\operatorname{gr}_\mathfrak{L}^*(\mathrm{M}))\subset\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{A}/\mathfrak{L})$. To see that $\mathrm{Z}=0$, it suffices by (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection~1, n\textsuperscript{o}~1, cor.~1 of prop.~2 and n\textsuperscript{o}~3, prop.~7) to see that for every $\mathfrak{q}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{A}/\mathfrak{L})$, $\mathrm{Z}_\mathfrak{q}=0$. But the ideals of $\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{A}/\mathfrak{L})$ are of the form $\mathfrak{p}/(\mathfrak{L}/\mathfrak{R})$, where $\mathfrak{p}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L})$, and it comes down to the same to see that $\mathrm{Z}_\mathfrak{p}=0$ for every $\mathfrak{p}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L})$. Now, if $\mathfrak{p}=\mathfrak{r}/\mathfrak{R}$, where $\mathfrak{r}$ is a prime ideal of $\mathrm{A}$, then $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))_\mathfrak{p}=\operatorname{gr}_{\mathfrak K_\mathfrak{r}}^*(\mathrm{M}_\mathfrak{r})$ and $(\mathrm{A}/\mathfrak{L})_\mathfrak{p}=\mathrm{A}_\mathfrak{r}/\mathfrak{L}_\mathfrak{r}$, and the images of the $f_i$ in $\mathrm{A}_\mathfrak{r}$ form a $\operatorname{gr}_{\mathfrak K_\mathfrak{r}}^*(\mathrm{A}_\mathfrak{r})$-regular sequence $(\mathbf{0},15.1.14)$; applying to $\mathrm{A}_\mathfrak{r}$, $\mathfrak K_\mathfrak{r}$ and $\mathrm{M}_\mathfrak{r}$ the implication $a)\Rightarrow b)$ of the statement, we see from the hypothesis $d)$ that $\psi_{\mathrm{M}_\mathfrak{r}}$ is bijective, hence also $\psi_{\mathrm{R}_\mathfrak{r}}'$ by virtue of \sref{IV.19.6.3}; in other words, $\mathrm{Q}_\mathfrak{r}=\mathrm{Q}_\mathfrak{r}'[\mathrm{T}_1,\ldots,\mathrm{T}_n]$, where we have set $\mathrm{Q}'=\operatorname{gr}_\mathfrak{R}^*(\mathrm{R},\mathrm{N})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{L})$, which is here a sub-$\mathrm{B}$-module of $\mathrm{P}$; in particular $\mathrm{Q}_\mathfrak{r}'=\mathrm{Q}_\mathfrak{r}\cap\mathrm{P}_\mathfrak{r}$; hence finally $\mathrm{Z}_\mathfrak{r}=\mathrm{Z}_\mathfrak{p}=0$, which completes the demonstration of \sref{IV.19.7.1.3}.

We will proceed by induction on $m$, assuming therefore \sref{IV.19.7.1.6} verified when $m$ is replaced by an integer $m'<m$. We will consider on the other hand, for $m$ fixed and for an integer $d\geqslant 0$, the relation
\begin{equation}
  \mathrm{R}\cap(\mathfrak{R}^m\mathrm{N}+\mathfrak{L}^{m+1}\mathrm{N})\subset(\mathrm{R}\cap\mathfrak{R}^d\mathrm{N}\cap\mathfrak{L}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N}.
  \tag{$*_d$}
\end{equation}

It is clear that it suffices to demonstrate $(*_d)$ for $d=m$, and on the other hand $(*_0)$ is trivially true. We will prove $(*_d)$ by \emph{induction on $d$}; in other words, we suppose, for a $d\geqslant 0$ fixed (with $d<m$), that $(*_d)$ is true, and we wish to prove
\begin{equation}
  \mathrm{R}\cap(\mathfrak{R}^m\mathrm{N}+\mathfrak{L}^{m+1}\mathrm{N})\subset(\mathrm{R}\cap\mathfrak{R}^{d+1}\mathrm{N}\cap\mathfrak{L}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N}.
  \tag{$*_{d+1}$}
\end{equation}

Consider for this, for an $h\geqslant 0$, the relation
\begin{equation}
  \mathrm{R}\cap(\mathfrak{R}^m\mathrm{N}+\mathfrak{R}^d\mathfrak{L}^h\mathrm{N})\subset(\mathrm{R}\cap(\mathfrak{R}^{d+1}\mathrm{N}+\mathfrak{R}^d\mathfrak{L}^h\mathrm{N})\cap\mathfrak{L}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N}
  \tag{$**_{d+1,h}$}
\end{equation}

The hypothesis $(*_d)$ implies that $(**_{d+1,0})$ is true. We will prove by \emph{induction on $h$} that $(**_{d+1,h})$ is true for every $h\geqslant 0$. But we have

\begin{lemma}[19.7.1.7]
\label{IV.19.7.1.7}
Let $\mathrm{A}$ be a Noetherian ring, $\mathrm{N}$ a finitely generated $\mathrm{A}$-module, $\mathfrak{L}$ an ideal of $\mathrm{A}$, $\mathrm{E}$, $\mathrm{F}$ two submodules of $\mathrm{N}$. For every $k>0$ there exists $h>0$ such that $\mathrm{E}\cap(\mathrm{F}+\mathfrak{L}^h\mathrm{N})\subset(\mathrm{E}\cap\mathrm{F})+\mathfrak{L}^k\mathrm{N}$.
\end{lemma}

\begin{proof}
Indeed, let $\varphi:\mathrm{N}\to\mathrm{N}/(\mathrm{E}\cap\mathrm{F})=\mathrm{N}_1$ be the canonical homomorphism, and set $\mathrm{E}_1=\varphi(\mathrm{E})$, $\mathrm{F}_1=\varphi(\mathrm{F})$, so that $\mathrm{E}_1\cap\mathrm{F}_1=0$ and $\varphi(\mathrm{E}\cap(\mathrm{F}+\mathfrak{L}^h\mathrm{N}))=\mathrm{E}_1\cap(\mathrm{F}_1+\mathfrak{L}^h\mathrm{N}_1)$.
We know $(\mathbf{0}_\mathrm{I},7.3.2)$ that there exists $h$ such that $(\mathrm{E}_1+\mathrm{F}_1)\cap\mathfrak{L}^h\mathrm{N}_1\subset\mathfrak{L}^k(\mathrm{E}_1+\mathrm{F}_1)$; if $h$ is thus chosen and if $x_1\in\mathrm{E}_1$ is such that there exists $y_1\in\mathrm{F}_1$ for which $x_1-y_1\in\mathfrak{L}^h\mathrm{N}_1$, we deduce $x_1-y_1\in\mathfrak{L}^k\mathrm{N}_1$, hence $\mathrm{E}_1\cap(\mathrm{F}_1+\mathfrak{L}^h\mathrm{N}_1)\subset\mathfrak{L}^k\mathrm{N}_1$; therefore $\mathrm{E}\cap(\mathrm{F}+\mathfrak{L}^h\mathrm{N})\subset\mathrm{E}\cap\mathrm{F}+\mathfrak{L}^k\mathrm{N}$, which proves the lemma.
\end{proof}

It suffices then to apply this lemma with $\mathrm{E}=\mathrm{R}\cap\mathfrak{L}^m\mathrm{N}$, $\mathrm{F}=\mathfrak{R}^{d+1}\mathrm{N}$ and to take $k=m+1$ to deduce that $(**_{d+1,h})$ implies $(*_{d+1})$.

We suppose therefore in all that follows that $(**_{d+1,h})$ \emph{is satisfied.}

\begin{remarks}[19.7.2]
\label{IV.19.7.2}
\medskip\noindent
\begin{enumerate}
\item[\rm(i)] When one supposes that $\operatorname{gr}_\mathfrak K^*(\mathrm{A})$ is an $(\mathrm{A}/\mathfrak K)$-module \emph{flat}, the hypothesis that the sequence $\mathbf{f}$ is $(\mathrm{A}/\mathfrak K)$-regular implies that it is also $\operatorname{gr}_\mathfrak K^*(\mathrm{A})$-regular $(\mathbf{0},15.1.14)$. We note that this will be the case when $\mathrm{A}$ and $\mathrm{A}/\mathfrak K$ \emph{are regular rings} $(\mathbf{0},17.3.6)$: indeed, $\mathfrak K$ is then a regular ideal of $\mathrm{A}$ \sref{IV.19.1.2}, hence quasi-regular, and localising at the maximal ideals of $\mathrm{A}$ containing $\mathfrak K$ and using $(\mathbf{0},15.1.7)$, one sees that $\operatorname{gr}_\mathfrak K^*(\mathrm{A})$ is a \emph{projective} $(\mathrm{A}/\mathfrak K)$-module.

\end{enumerate}

\begin{corollary}[19.7.3]
\label{IV.19.7.3}
Let $\mathrm{A}$ be a local Noetherian ring, $\mathfrak{m}$ its maximal ideal, $\mathfrak K$ an ideal of $\mathrm{A}$ such that the ring $\mathrm{A}/\mathfrak K$ is regular and of dimension $n$, $\mathrm{M}$ a finitely generated $\mathrm{A}$-module. The following conditions are equivalent:
\begin{enumerate}
\item[\rm a)] $\operatorname{gr}_\mathfrak K^*(\mathrm{M})$ is a flat $(\mathrm{A}/\mathfrak K)$-module.
\item[\rm b)] If $\mathrm{P}(\mathrm{T})$ and $\mathrm{Q}(\mathrm{T})$ are the Poincar\'e series of the $(\mathrm{A}/\mathfrak{m})$-modules graded $\operatorname{gr}_\mathfrak{m}^*(\mathrm{M})$ and $\operatorname{gr}_\mathfrak K^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{m})$, we have
\end{enumerate}
\begin{equation}
  \mathrm{Q}(\mathrm{T})=\mathrm{P}(\mathrm{T})(1-\mathrm{T})^n.
  \tag{19.7.3.1}
\label{IV.19.7.3.1}
\end{equation}

Let $\mathbf{f}=(f_1,\ldots,f_n)$ be a sequence of elements of $\mathrm{A}$ such that the images of the $f_i$ in $\mathrm{A}/\mathfrak K$ form a regular system of parameters of $\mathrm{A}/\mathfrak K$ $(\mathbf{0},17.1.6)$, so that $\mathbf{f}$ is an $(\mathrm{A}/\mathfrak K)$-regular sequence; moreover, if $\mathfrak{J}$ is the ideal generated by $\mathbf{f}$, $\mathfrak{J}+\mathfrak K=\mathfrak{m}$, and as $\mathrm{A}/\mathfrak{m}$ is a field, $\operatorname{gr}_\mathfrak{m}^*(\mathrm{M})$ is a flat $(\mathrm{A}/\mathfrak{m})$-module. As $\mathrm{A}$ is Noetherian and $\mathrm{M}$ a finitely generated $\mathrm{A}$-module, one can apply the equivalence of $a)$ and $b)$ in \sref{IV.19.7.1}. Moreover, $\mathrm{A}/\mathfrak{m}$ being a field, the fact that $\psi_\mathrm{M}$ is bijective is equivalent to saying that for every $h\geqslant 0$, $\operatorname{gr}_\mathfrak{m}^h(\mathrm{M})$ and the submodule of elements of degree $h$ of $(\operatorname{gr}_\mathfrak K^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{m}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ have the same rank over $\mathrm{A}/\mathfrak{m}$; but for the second of these modules the rank in question is clearly equal to
\[
  \textstyle\sum_{r=0}^{h}\binom{r+n-1}{n-1}\operatorname{rg}(\operatorname{gr}_\mathfrak K^{h-r}(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak K}(\mathrm{A}/\mathfrak{m})).\quad\text{As}\quad(1-\mathrm{T})^{-n}=\textstyle\sum_{r=0}^{\infty}\binom{r+n-1}{n-1}\mathrm{T}^r,\quad\text{this}
\]
proves that the relation \sref{IV.19.7.3.1} is necessary and sufficient for $\psi_\mathrm{M}$ to be bijective.
\end{corollary}

\begin{corollary}[19.7.5]
\label{IV.19.7.5}
Let $\mathrm{X}$ be a locally Noetherian prescheme, $\mathrm{Y}$ a closed sub-prescheme of $\mathrm{X}$, $\mathrm{Z}$ a closed sub-prescheme of $\mathrm{Y}$; one supposes $\mathrm{Y}$ and $\mathrm{Z}$ regular. Let $\sh{F}$ be a coherent $\sh{O}_\mathrm{X}$-Module.
\begin{enumerate}
\item[\rm(i)] If $\sh{F}$ is normally flat along $\mathrm{Y}$ at the points of $\mathrm{Z}$ (cf. \sref{IV.11.3.4} for the definition of normal flatness at a point), then $\sh{F}$ is normally flat along $\mathrm{Z}$.
\item[\rm(ii)] Suppose moreover that $\mathrm{Z}$ is connected and that $\sh{O}_\mathrm{X}$ is normally flat along $\mathrm{Y}$ at the points of $\mathrm{Z}$ (which is the case in particular if $\mathrm{X}$ is regular at the points of $\mathrm{Z}$). If $\sh{F}$ is normally flat along $\mathrm{Z}$ and if there exists a point $z\in\mathrm{Z}$ at which $\sh{F}$ is normally flat along $\mathrm{Y}$, then $\sh{F}$ is normally flat along $\mathrm{Y}$ at all the points of $\mathrm{Z}$.
\end{enumerate}

Let $\sh{K}$ and $\sh{L}\supset\sh{K}$ be the coherent Ideals of $\sh{O}_\mathrm{X}$ defining respectively $\mathrm{Y}$ and $\mathrm{Z}$. For every point $z\in\mathrm{Z}$, the canonical immersion $\mathrm{Z}\to\mathrm{Y}$ is regular at $z$, since $\mathrm{Y}$ and $\mathrm{Z}$ are regular at that point \sref{IV.19.1.1}; in other words, there exists a sequence $(f_i)_{1\leqslant i\leqslant n}$ of elements of $\sh{O}_{\mathrm{X},z}$ which is $\sh{K}_z$-regular and such that $\sh{L}_z=\sh{K}_z+\sum_i f_i\sh{O}_{\mathrm{X},z}$.
\begin{enumerate}
\item[\rm(i)] The hypothesis means that $\operatorname{gr}_{\sh{K}_z}^\bullet(\sh{F}_z)$ is a flat $\sh{O}_{\mathrm{Y},z}$-module; it therefore implies that $\operatorname{gr}_{\sh{L}_z}^\bullet(\sh{F}_z)$ is a flat $\sh{O}_{\mathrm{Z},z}$-module, by the implication $a)\Rightarrow b)$ of \sref{IV.19.7.1}.
\item[\rm(ii)] The supplementary hypothesis on $\mathrm{X}$ implies that the sequence $(f_i)$ is $\operatorname{gr}_{\sh{K}_z}^\bullet(\sh{O}_{\mathrm{X},z})$-regular at every point $z\in\mathrm{Z}$ \sref{IV.19.7.2}. On the other hand, $\mathrm{Z}$, being regular and connected, is integral; if $\operatorname{gr}_{\sh{K}_z}^\bullet(\sh{F}_z)$ is flat over $\sh{O}_{\mathrm{Y},z}$ at one point $z\in\mathrm{Z}$, then, since the generic point $\zeta$ of $\mathrm{Z}$ is a generization of $z$, $\operatorname{gr}_{\sh{K}_\zeta}^\bullet(\sh{F}_\zeta)$ is flat over $\sh{O}_{\mathrm{Y},\zeta}$ $(\mathbf{0}_\mathrm{I},6.3.1)$. \end{enumerate}
\end{corollary}

\begin{lemma}[19.9.2]
\label{IV.19.9.2}
Let $\mathrm{A}$ be a Noetherian ring, $\mathrm{M}$ a finitely generated $\mathrm{A}$-module, $\mathfrak{J}$ an ideal of $\mathrm{A}$. In order that, for every prime ideal $\mathfrak{p}\supset\mathfrak{J}$, we have $\operatorname{prof}_{\mathrm{A}_\mathfrak{p}}(\mathrm{M}_\mathfrak{p})\geqslant r$, it is necessary and sufficient that there exist an M-regular sequence of $r$ elements belonging to $\mathfrak{J}$.
\end{lemma}

\begin{proof}
The sufficiency of the condition follows immediately from $(\mathbf{0},16.4.5)$; let us prove its necessity. We argue by induction on $r$ (there is nothing to prove if $r=0$): since $\operatorname{prof}_{\mathrm{A}_\mathfrak{p}}(\mathrm{M}_\mathfrak{p})\geqslant r\setmin 1$ for every $\mathfrak{p}\supset\mathfrak{J}$, there exists by the induction hypothesis an M-regular sequence $(g_i)_{1\leqslant i\leqslant r-1}$ of elements of $\mathfrak{J}$. Set $\mathrm{N}=\mathrm{M}/(g_1\mathrm{M}+\ldots+g_{r-1}\mathrm{M})$; for every $\mathfrak{p}\supset\mathfrak{J}$, the images of the $g_i$ in $\mathrm{A}_\mathfrak{p}$ belong to the maximal ideal $\mathfrak{p}\mathrm{A}_\mathfrak{p}$ and form an $\mathrm{M}_\mathfrak{p}$-regular sequence $(\mathbf{0},15.1.14)$. It follows from $(\mathbf{0},16.4.6)$ that $\operatorname{prof}_{\mathrm{A}_\mathfrak{p}}(\mathrm{N}_\mathfrak{p})=\operatorname{prof}_{\mathrm{A}_\mathfrak{p}}(\mathrm{M}_\mathfrak{p})\setmin(r\setmin 1)\geqslant 1$, so everything reduces to proving the lemma for $r=1$. The hypothesis then means that, for every $\mathfrak{p}\supset\mathfrak{J}$, $\mathfrak{p}\mathrm{A}_\mathfrak{p}$ is not associated with $\mathrm{M}_\mathfrak{p}$ $(\mathbf{0},16.4.6)$; hence (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection~1, n\textsuperscript{o}~2, prop.~5), $\mathfrak{p}$ is not associated with $\mathrm{M}$. In other words, $\mathfrak{J}$ is contained in none of the prime ideals of $\operatorname{Ass}(\mathrm{M})$, and therefore is not contained in their union (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection~1, n\textsuperscript{o}~1, prop.~2). Since this union is the set of elements that are not M-regular (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection~1, cor.~2 of prop.~2), this proves the lemma.
\end{proof}

\begin{proposition}[19.9.5]

\begin{proposition}[19.9.6]
\label{IV.19.9.6}
Let $f:\mathrm{X}\to\mathrm{S}$ be a morphism locally of finite presentation, $\mathrm{Y}$ a closed subset of $\mathrm{X}$, $\sh{F}$ a coherent $\sh{O}_\mathrm{X}$-Module of finite presentation and $f$-flat, $n$ an integer $>0$. For every point $y\in\mathrm{Y}$, denoting by $\mathrm{X}_{f(y)}$ the fibre of $f$ at the point $f(y)$, by $\sh{F}_{f(y)}$ the inverse image of $\sh{F}$ in $\mathrm{X}_{f(y)}$, and setting $\mathrm{Y}_{f(y)}=\mathrm{X}_{f(y)}\cap\mathrm{Y}$:
\begin{enumerate}
\item[\rm a)] $\operatorname{prof}_{\mathrm{Y}_{f(y)},y}(\sh{F}_{f(y)})\geqslant n$.
\item[\rm b)] There exists an open neighbourhood $\mathrm{U}$ of $y$ in $\mathrm{X}$ and a sequence $(g_i)_{1\leqslant i\leqslant n}$ of $n$ sections of $\sh{O}_\mathrm{X}$ above $\mathrm{U}$, that is transversally $\sh{F}$-regular relative to $\mathrm{S}$ (cf. \sref{IV.19.2.1} for the terminology) and such that $g_i(x)=0$ for $1\leqslant i\leqslant n$ at every point $x\in\mathrm{U}\cap\mathrm{Y}$.
\end{enumerate}

\begin{proposition}[19.9.8]
\label{IV.19.9.8}
Let $f:X\to S$ be a morphism locally of finite presentation, $Z$ a closed subset of $X$, $\sh{F}$ an $\sh{O}_X$-module of finite presentation and $f$-flat, and $\sh{G}$ a quasi-coherent $\sh{O}_S$-module.
Suppose that, for every $x\in Z$, we have $\operatorname{prof}((\sh{F}_{f(x)})_x)\geqslant 1$ (resp.\ $\operatorname{prof}((\sh{F}_{f(x)})_x)\geqslant 2$).
Then, if $j:X\setmin Z\to X$ is the canonical injection and if we set $\sh{H}=\sh{F}\otimes_{\sh{O}_X}f^*(\sh{G})$, the canonical homomorphism $\rho_{\sh{H}}:\sh{H}\to j_*(j^*(\sh{H}))$ relative to $j$ is injective (resp.\ bijective).
\end{proposition}

\begin{proof}
The question is evidently local on $X$ and $S$, and it suffices to prove the proposition in a neighbourhood of a point $x\in Z$. By \sref{IV.19.9.6}, we may suppose that there exists a section $g_1$ of $\sh{O}_X$ over $X$ that is transversally $\sh{F}$-regular relative to $S$ (resp.\ a transversally $\sh{F}$-regular sequence $(g_1,g_2)$ of two sections of $\sh{O}_X$ over $X$ relative to $S$) such that, if $Z'$ is the set of $x'\in X$ for which $g_1(x')=0$ (resp.\ $g_1(x')=g_2(x')=0$), then $Z\subset Z'$.

This being so, return to the proof of \sref{IV.19.9.8}. By \sref{IV.19.9.6}, we still have $\operatorname{prof}((\sh{F}_{f(x)})_x)\geqslant 1$ (resp.\ $\operatorname{prof}((\sh{F}_{f(x)})_x)\geqslant 2$) for every $x\in Z'$.
